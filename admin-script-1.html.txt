    function callServer(method) {
      const args = Array.prototype.slice.call(arguments, 1);
      return new Promise(function (resolve, reject) {
        if (!window.google || !google.script || !google.script.run) {
          reject(new Error("Apps Script 실행 환경에서 열어주세요."));
          return;
        }
        const runner = google.script.run
          .withSuccessHandler(resolve)
          .withFailureHandler(function (error) {
            const message = error && error.message ? error.message : String(error || "요청을 처리하지 못했습니다.");
            reject(new Error(message));
          });
        runner[method].apply(runner, args);
      });
    }

    function api(method) {
      const args = [method, STATE.token].concat(Array.prototype.slice.call(arguments, 1));
      return callServer.apply(null, args).catch(function (error) {

          clearSession();
          showLogin();
          showToast("로그인이 필요합니다.", error.message, "error");
        }
        throw error;
      });
    }

    function reportLoginPhaseError(error) {

      if (!isAppVisible()) {
        showLoginError(message);
        if (ELEMENTS.loginPassword) ELEMENTS.loginPassword.focus();

      renderFatalError("로그인 후 화면을 여는 중 문제가 발생했습니다.", message);
      try { showToast("화면 표시 오류", message, "error"); } catch (toastError) {}
    }

    /** 어떤 이유로든 화면이 비어버리면, 그 이유를 화면에 띄웁니다. */
    function installGlobalErrorNet() {
      function handle(message) {
        if (!isAppVisible() || !isContentEmpty()) return;
        renderFatalError("화면을 그리는 중 문제가 발생했습니다.", message);
      }
      window.addEventListener("error", function (event) {
        handle((event && event.message) || "스크립트 오류");
      });
      window.addEventListener("unhandledrejection", function (event) {
        var reason = event && event.reason;
        handle((reason && reason.message) || String(reason || "처리되지 않은 오류"));
      });
    }

    function handleLogin(event) {
      event.preventDefault();
      if (STATE.busy) return;
      const id = ELEMENTS.loginId.value.trim();
      const password = ELEMENTS.loginPassword.value;
      ELEMENTS.loginError.classList.add("hidden");
      if (!id || !password) {
        showLoginError("아이디와 비밀번호를 모두 입력해주세요.");
        return;
      }
      setLoginBusy(true);
      callServer("portalLogin", { id: id, password: password })
        .then(function (result) {
          STATE.token = result.token;
          persistSession(result.expiresAt);
          applyBootstrap(result);
          ELEMENTS.loginPassword.value = "";
          showApp();
          if (result.mustReplaceInitialPassword) {
            showChangePasswordModal(true);
            return Promise.resolve();
          }
          var requestedCourseId = getRequestedCourseId();
          if (requestedCourseId) return openCourse(requestedCourseId);
          return navigate("dashboard");
        })
        .catch(function (error) {
          reportLoginPhaseError(error);
        })
        .finally(function () {
          setLoginBusy(false);
        });
    }

    function pushLoginGuardState() {
      try {
        if (window.google && google.script && google.script.history) {
          google.script.history.push({ vupLoginGuard: true }, null, "login");
          return;
        }
        window.history.pushState(
          { vupLoginGuard: true },
          "",
          window.location.pathname + window.location.search + "#login"
        );
      } catch (error) {}
    }

    /** 로그인 화면에서 발생한 뒤로가기인지 판단해 처리합니다. 처리했으면 true. */
    function handleLoginExitAttempt() {
      if (!STATE.loginExitGuard) return false;
      if (ELEMENTS.loginView.classList.contains("hidden")) return false;
      const leave = window.confirm(
        "해당 페이지를 나가겠습니까?\n로그인 화면을 벗어나면 처음부터 다시 접속해야 합니다."
      );
      if (leave) {
        STATE.loginExitGuard = false;
        try { window.history.back(); } catch (error) {}
        return true;
      }
      pushLoginGuardState();
      return true;
    }

    function initializePortalHistory() {
      const baseUrl = window.location.pathname + window.location.search;
      STATE.portalHistoryKey = "dashboard||";
      if (
        window.google &&
        google.script &&
        google.script.history
      ) {
        google.script.history.replace(
          { vupPortalGuard: true },
          null,
          "portal"
        );
        google.script.history.push(
          { vupPortal: true, view: "dashboard", courseId: "", tab: "" },
          null,
          "dashboard"
        );
        return;
      }
      window.history.replaceState(
        { vupPortalGuard: true },
        "",
        baseUrl + "#portal"
      );
      window.history.pushState(
        { vupPortal: true, view: "dashboard", courseId: "", tab: "" },
        "",
        baseUrl + "#dashboard"
      );
    }

    function writePortalHistory(view, options) {
      const settings = options || {};
      const state = {
        vupPortal: true,
        view: String(view || "dashboard"),
        courseId: String(settings.courseId || ""),
        tab: String(settings.tab || "")
      };
      const routeKey = state.view + "|" + state.courseId + "|" + state.tab;
      const sameLocation = STATE.portalHistoryKey === routeKey;
      STATE.portalHistoryKey = routeKey;
      const hash = buildPortalHistoryHash(state.view, state.courseId, state.tab);
      if (
        window.google &&
        google.script &&
        google.script.history
      ) {
        if (settings.replace || sameLocation) {
          google.script.history.replace(state, null, hash);
        } else {
          google.script.history.push(state, null, hash);
        }
        return;
      }
      const current = window.history.state || {};
      const sameWindowLocation =
        current.vupPortal === true &&
        current.view === state.view &&
        String(current.courseId || "") === state.courseId &&
        String(current.tab || "") === state.tab;
      const url = buildPortalHistoryUrl(state.view, state.courseId, state.tab);
      if (settings.replace || sameWindowLocation) {
        window.history.replaceState(state, "", url);
      } else {
        window.history.pushState(state, "", url);
      }
    }

    function buildPortalHistoryHash(view, courseId, tab) {
      if (view === "course" && courseId) {
        return "course/" + encodeURIComponent(courseId) +
          (tab ? "/" + encodeURIComponent(tab) : "");
      }
      return encodeURIComponent(view || "dashboard");
    }

    function buildPortalHistoryUrl(view, courseId, tab) {
      const baseUrl = window.location.pathname + window.location.search;
      if (view === "course" && courseId) {
        return baseUrl + "#course/" + encodeURIComponent(courseId) +
          (tab ? "/" + encodeURIComponent(tab) : "");
      }
      return baseUrl + "#" + encodeURIComponent(view || "dashboard");

    function handlePortalPopState(event) {
      if (handleLoginExitAttempt()) return;
      if (!STATE.token || ELEMENTS.appView.classList.contains("hidden")) return;
      const historyState = event.state || {};
      const incomingKey =
        String(historyState.view || "") + "|" +
        String(historyState.courseId || "") + "|" +
        String(historyState.tab || "");
      if (
        historyState.vupPortal &&
        incomingKey === STATE.portalHistoryKey &&
        String(historyState.view || "") === String(STATE.view || "") &&
        (
          historyState.view !== "course" ||
          (
            STATE.course &&
            String(STATE.course.siteId || "") === String(historyState.courseId || "") &&
            String(STATE.currentCourseTab || "overview") ===
              String(historyState.tab || "overview")
          )
        )
      ) {
        return;
      }
      if (historyState.vupPortalGuard) {
        STATE.portalHistoryKey = "";
        writePortalHistory("dashboard");
        navigate("dashboard", { fromHistory: true });
        return;
      }
      if (
        historyState.vupPortal &&
        historyState.view === "course" &&
        historyState.courseId
      ) {
        STATE.portalHistoryKey =
          "course|" + String(historyState.courseId || "") + "|" +
          String(historyState.tab || "overview");
        openCourse(
          historyState.courseId,
          historyState.tab || "overview",
          { fromHistory: true }
        );
        return;
      }
      if (historyState.vupPortal && historyState.view) {
        STATE.portalHistoryKey =
          String(historyState.view) + "||";
        navigate(historyState.view, { fromHistory: true });
        return;
      }
      STATE.portalHistoryKey = "";
      writePortalHistory("dashboard");
      navigate("dashboard", { fromHistory: true });
    }

    function handlePortalHistoryChange(event) {
      handlePortalPopState({ state: event && event.state ? event.state : {} });
    }

    function persistSession(expiresAt) {
      STATE.sessionExpiresAt = new Date(expiresAt || 0).getTime();
      try {
        sessionStorage.setItem(SESSION_KEY, JSON.stringify({
          token: STATE.token,
          expiresAt: STATE.sessionExpiresAt
        }));
      } catch (error) {}
      startIdleGuard();
    }

    function restoreStoredSession() {
      let stored = null;
      try { stored = JSON.parse(sessionStorage.getItem(SESSION_KEY) || "null"); }
      catch (error) { stored = null; }
      if (!stored || !stored.token || Number(stored.expiresAt || 0) <= Date.now()) {
        clearStoredSession();
        showLogin();
        return;
      }
      STATE.token = String(stored.token);
      STATE.sessionExpiresAt = Number(stored.expiresAt || 0);
      showLoader("로그인 복구 중", "안전한 세션을 확인하고 있습니다.");
      callServer("portalRestoreSession", STATE.token)
        .then(function (result) {
          applyBootstrap(result);
          showApp();
          startIdleGuard();
          if (result.mustReplaceInitialPassword) {
            showChangePasswordModal(true);
            return Promise.resolve();
          }
          var restoredCourseId = getRequestedCourseId();
          return restoredCourseId ? openCourse(restoredCourseId) : navigate("dashboard");
        })
        .catch(function () {
          clearSession();
          showLogin();
        })
        .finally(hideLoader);
    }

    function recordActivity() {
      STATE.lastActivityAt = Date.now();
    }

    function startIdleGuard() {
      recordActivity();
      if (STATE.idleTimer) window.clearInterval(STATE.idleTimer);
      STATE.idleTimer = window.setInterval(function () {
        if (!STATE.token) return;
        if (Date.now() >= Number(STATE.sessionExpiresAt || 0) ||
            Date.now() - STATE.lastActivityAt >= CLIENT_IDLE_MS) {
          handleLogout();
          showToast("로그인이 만료되었습니다.", "보안을 위해 다시 로그인해주세요.", "warning");
        }
      }, 30000);
    }

    function navigate(view, navigationOptions) {
      const target = String(view || "dashboard");
      const navigation = navigationOptions || {};
      if (STATE.view === "course" && STATE.pageBuilderDirty && target !== "course" && !navigation.skipPageBuilderGuard) {
        showModal(
          "저장하지 않은 사용자 화면 변경이 있습니다",
          "지금 이동하면 방금 추가하거나 수정한 카드가 사라집니다.",
          '<div class="warning-callout"><strong>실제 사용자 사이트에는 아직 반영되지 않았습니다.</strong><span>취소한 뒤 ‘변경사항 실제 사이트에 반영’을 먼저 눌러주세요.</span></div>',
          '<button type="button" class="btn btn-secondary" data-modal-close>계속 편집</button><button type="button" class="btn btn-danger" data-discard-page-builder="' + escapeAttr(target) + '">변경 버리고 이동</button>'
        );
        return Promise.resolve(false);
      }
      const requestVersion = ++STATE.navigationVersion;
      if (target === "developer-inquiries" && !(STATE.admin && STATE.admin.isDeveloper)) {
        return Promise.resolve();
      }
      if (!navigation.fromHistory) {
        writePortalHistory(target, {
          replace: Boolean(navigation.replaceHistory)
        });
      }
      STATE.view = target;
      STATE.course = target === "course" ? STATE.course : null;
      updateNavigationChrome(target);
      closeSidebar();
      ELEMENTS.contentRoot.scrollTop = 0;
      window.scrollTo({ top: 0, behavior: "auto" });

      if (target === "dashboard") return refreshDashboard(requestVersion, Boolean(navigation.forceRefresh));
      if (target === "profile") {
        return ensureProfileLoaded().then(function () {
          if (STATE.view === "profile") renderProfile();
        });
      }
      if (target === "create") {
        // 담당자 이름·연락처를 프로필에서 자동 채우므로 먼저 확보합니다.
        return ensureProfileLoaded().then(function () {
          if (STATE.view === "create") renderCreateCourse();
        });
      }
      if (target === "courses") return loadCourseList("all", requestVersion);
      if (target === "history") return loadCourseList("history", requestVersion);
      if (target === "satisfaction") return renderSatisfaction(requestVersion);
      if (target === "inquiries") return loadInquiries(false, requestVersion);
      if (target === "trash") return loadTrash(requestVersion);
      if (target === "developer-inquiries") return loadInquiries(true, requestVersion);
      return Promise.resolve();
    }

    function updateNavigationChrome(view) {
      const displayName =
        (STATE.profile && STATE.profile.name) ||
        (STATE.admin && (STATE.admin.displayName || STATE.admin.id)) ||
        "관리자";
      const meta = VIEW_META[view] || VIEW_META.dashboard;
      ELEMENTS.topbarEyebrow.textContent = meta.eyebrow;
      ELEMENTS.topbarTitle.textContent = view === "dashboard"
        ? displayName + "님의 " + meta.title
        : meta.title;
      document.querySelectorAll(".nav-item").forEach(function (item) {
        const active = item.dataset.view === view;
        item.classList.toggle("active", active);
        if (active) item.setAttribute("aria-current", "page");
        else item.removeAttribute("aria-current");
      });
    }

    function refreshDashboard(requestVersion, forceRefresh) {
      renderDashboard(STATE.dashboard || {});
      if (
        !forceRefresh &&
        STATE.dashboardFetchedAt &&
        Date.now() - STATE.dashboardFetchedAt < 15000
      ) {
        return Promise.resolve(STATE.dashboard);
      }
      return api("portalGetDashboard")
        .then(function (dashboard) {
          if (
            requestVersion !== STATE.navigationVersion ||
            STATE.view !== "dashboard"
          ) return dashboard;
          STATE.dashboard = dashboard;
          STATE.dashboardFetchedAt = Date.now();
          if (dashboard.profile) {
            STATE.profile = dashboard.profile;
            STATE.profileLoaded = true;
          }
          updateAccountChrome();
          renderDashboard(dashboard);
          return dashboard;
        })
        .catch(function (error) {
          if (
            requestVersion !== STATE.navigationVersion ||
            STATE.view !== "dashboard"
          ) return;
          showToast("대시보드를 새로고침하지 못했습니다.", error.message, "error");
        });
    }

    function loadCourseList(filter, navigationVersion) {
      const requestedFilter = String(filter || "all");
      const expectedView = STATE.view === "history" ? "history" : "courses";
      const requestNavigationVersion =
        Number(navigationVersion || STATE.navigationVersion);
      const listRequestVersion = ++STATE.listRequestVersion;

      STATE.listFilter = requestedFilter === "history"
        ? "history"
        : ["all", "manage", "blocked"].indexOf(requestedFilter) !== -1
          ? requestedFilter
          : "all";

      STATE.courseDeleteMode = false;
      STATE.selectedCourseIds = [];

      const cacheKey = [
        expectedView,
        STATE.listFilter,
        STATE.listSearch
      ].join("|");
      const cachedCourses = STATE.courseListCache[cacheKey];
      if (cachedCourses) {
        STATE.courses = cachedCourses.slice();
        renderCourseList(expectedView === "history" ? "history" : "manage");
      } else {
        renderCourseListLoading(STATE.listFilter);
      }

      return api("portalListCourses", {
        filter: STATE.listFilter,
        scope: expectedView === "history" ? "history" : "operations",
        search: STATE.listSearch
      })
        .then(function (result) {
          if (
            listRequestVersion !== STATE.listRequestVersion ||
            requestNavigationVersion !== STATE.navigationVersion ||
            STATE.view !== expectedView
          ) return result;
          STATE.courses = result.courses || [];
          STATE.courseListCache[cacheKey] = STATE.courses.slice();
          renderCourseList(expectedView === "history" ? "history" : "manage");
          return result;
        })
        .catch(function (error) {
          if (
            listRequestVersion !== STATE.listRequestVersion ||
            requestNavigationVersion !== STATE.navigationVersion ||
            STATE.view !== expectedView
          ) return;
          renderPageError(
            "강의 목록을 불러오지 못했습니다.",
            error.message
          );
        });
    }

    /* 지표 카드 아래쪽에 해당 구분의 교육명을 최대 2줄까지 보여줍니다.
       보여줄 교육이 없는 줄은 첫 줄에만 회색 안내문을 두고,
       나머지 줄은 카드 높이를 유지하기 위한 빈 줄로 남깁니다. */
    function metricCardTitleList(titles) {
      const list = (Array.isArray(titles) ? titles : [])
        .filter(function (title) { return String(title || "").trim(); })
        .slice(0, METRIC_TITLE_ROWS);
      let rows = "";

      for (let index = 0; index < METRIC_TITLE_ROWS; index++) {
        if (index < list.length) {
          rows += '<li class="metric-title-item">' +
            '<i class="metric-title-dot" aria-hidden="true"></i>' +
            '<span>' + escapeHtml(list[index]) + '</span></li>';
        } else if (index === list.length) {
          rows += '<li class="metric-title-item is-empty">' +
            '<i class="metric-title-dot" aria-hidden="true"></i>' +
            '<span>해당란에 표시될 교육이 없습니다.</span></li>';
        } else {
          rows += '<li class="metric-title-item is-blank" aria-hidden="true">' +
            '<i class="metric-title-dot" aria-hidden="true"></i><span></span></li>';
        }
      }

      return '<ul class="metric-title-list">' + rows + '</ul>';
    }

    function handleContentClick(event) {
      // 설정 묶음 접기/펴기
      const settingGroupHead = event.target.closest("[data-setting-group]");
      if (settingGroupHead) {
        const key = settingGroupHead.dataset.settingGroup;
        STATE.settingGroupOpen[key] = STATE.settingGroupOpen[key] === false;
        updateCourseTabContent();
        return;
      }

      // 자리 배정 단계 이동
      const stepButton = event.target.closest("[data-seating-step]");
      if (stepButton) {
        STATE.seatingStep = stepButton.dataset.seatingStep;
        updateCourseTabContent();
        return;
      }

      // 자동 / 수동 선택
      const modeButton = event.target.closest("[data-seating-mode]");
      if (modeButton) {
        const parts = String(modeButton.dataset.seatingMode).split(":");
        STATE.seatingMode[parts[0]] = parts[1] || "";
        updateCourseTabContent();
        return;
      }

      // 명부 필터
      const rosterFilterTab = event.target.closest("[data-roster-filter]");
      if (rosterFilterTab) {
        STATE.rosterFilter = rosterFilterTab.dataset.rosterFilter;
        updateCourseTabContent();
        return;
      }

      // 좌석표 / 방 배정 편집 (다른 처리보다 먼저)
      const seatCell = event.target.closest("[data-seat]");
      if (seatCell) { handleSeatCellClick(seatCell); return; }
      const roomPerson = event.target.closest("[data-room-person]");
      if (roomPerson) { handleRoomPersonClick(roomPerson); return; }

      // 강의 복제 (data-course 처리보다 먼저 가로챕니다)
      const duplicate = event.target.closest("[data-duplicate-course]");
      if (duplicate) {
        event.preventDefault();
        event.stopPropagation();
        openDuplicateCourseModal(duplicate.dataset.duplicateCourse);
        return;
      }
      const courseButton = event.target.closest("[data-course]");
      if (courseButton) {
        openCourse(courseButton.dataset.course, courseButton.dataset.openCourseTab || "overview");
        return;
      }

      const viewButton = event.target.closest("[data-view]");
      if (viewButton) {
        navigate(viewButton.dataset.view);
        return;
      }

      const tabButton = event.target.closest("[data-course-tab]");
      if (tabButton) {
        STATE.currentCourseTab = tabButton.dataset.courseTab;
        if (STATE.course && STATE.course.siteId) {
          writePortalHistory("course", {
            courseId: STATE.course.siteId,
            tab: STATE.currentCourseTab
          });
        }
        updateCourseTabContent();
        loadCourseTabResources(STATE.currentCourseTab);
        return;
      }

      const filterButton = event.target.closest("[data-list-filter]");
      if (filterButton) {
        STATE.listFilter = filterButton.dataset.listFilter;
        loadCourseList(STATE.listFilter, STATE.navigationVersion);
        return;
      }

      const satisfactionTab = event.target.closest("[data-satisfaction-tab]");
      if (satisfactionTab) {
        STATE.satisfactionTab = satisfactionTab.dataset.satisfactionTab;
        renderSatisfactionTab();
        return;
      }

      const surveyTypeFilter = event.target.closest("[data-survey-type-filter]");
      if (surveyTypeFilter) {
        STATE.satisfactionFilter.formType = surveyTypeFilter.dataset.surveyTypeFilter;
        loadSatisfactionList(false);
        return;
      }

      const analysisMode = event.target.closest("[data-survey-analysis-mode]");
      if (analysisMode) {
        STATE.satisfactionAnalysisMode = analysisMode.dataset.surveyAnalysisMode;
        renderSatisfactionAnalysisTab();
        return;
      }

      const analyzeButton = event.target.closest("[data-survey-analyze]");
      if (analyzeButton) {
        loadSatisfactionAnalysis(analyzeButton.dataset.surveyAnalyze);
        return;
      }

      const action = event.target.closest("[data-action]");
      if (action) handleAction(action.dataset.action, action);

      const deleteButton = event.target.closest("[data-delete-file]");
      if (deleteButton) {
        confirmDeleteFile(
          deleteButton.dataset.deleteFile,
          deleteButton.dataset.fileType,
          deleteButton.dataset.fileName
        );
      }
    }

    function handleContentChange(event) {
      if (event.target.id === "operationCurrentStage") {
        ensureOperationsPlan().stage = event.target.value;
        persistOperationsPlan("현재 운영 단계를 변경했습니다.");
        return;
      }
      if (event.target.closest("#courseCreateForm")) {
        scheduleCreateDraftSave();
        const draftStatus = document.getElementById("createDraftStatus");
        if (draftStatus) draftStatus.innerHTML = "<i></i> 저장 중…";
        updateCreateSummary();
      }
      if (event.target.name === "coursePrimaryContact") {
        updateCreateSummary();
        showToast("대표 문의 담당자를 변경했습니다.", "한 명만 선택할 수 있으며 사용자 페이지 최상단에 노출됩니다.");
        return;
      }
      if (event.target.classList.contains("meal-checkbox") && event.target.closest("#courseCreateForm")) {
        updateMealSelectionSummary();
        updateCreateSummary();
      }

      // --- 자리 배치: 포함/제외 토글 ---
      const seatingToggle = event.target.closest("[data-seating-toggle]");
      if (seatingToggle) {
        const excluded = !seatingToggle.checked;
        const text = seatingToggle.parentElement.querySelector(".switch-text");
        if (text) text.textContent = excluded ? "제외" : "포함";
        saveSeatingFlag(seatingToggle.dataset.seatingToggle, { excluded: excluded }, true);
        return;
      }

      // --- 교육 구성 편집 ---
      const segmentField = event.target.closest("[data-segment-field]");
      if (segmentField) {
        const index = Number(segmentField.dataset.segmentIndex);
        const field = segmentField.dataset.segmentField;
        const segment = (STATE.segments || [])[index];
        if (segment) {
          segment[field] = segmentField.value;
          if (field === "startDate" && (!segment.endDate || segment.endDate < segment.startDate)) {
            segment.endDate = segment.startDate;
          }
          if (field === "type" && segment.type === "online") segment.meals = [];
          updateCourseTabContent();
        }
        return;
      }
      const segmentMeal = event.target.closest("[data-segment-meal]");
      if (segmentMeal) {
        const parts = String(segmentMeal.dataset.segmentMeal).split(":");
        const segment = (STATE.segments || [])[Number(parts[0])];
        if (segment) {
          const day = Number(parts[1]);
          const type = parts[2];
          segment.meals = (segment.meals || []).filter(function (meal) {
            return !(Number(meal.day) === day && meal.type === type);
          });
          if (segmentMeal.checked) segment.meals.push({ day: day, type: type });
          updateCourseTabContent();
        }
        return;
      }

      // --- 명부: 성별 ---
      const genderSelect = event.target.closest("[data-roster-gender]");
      if (genderSelect) {
        saveSeatingFlag(genderSelect.dataset.rosterGender, { gender: genderSelect.value }, true);
        return;
      }

      // --- 명부: 생년 ---
      const birthInput = event.target.closest("[data-roster-birth]");
      if (birthInput) {
        const value = String(birthInput.value || "").replace(/[^\d]/g, "").slice(0, 4);
        birthInput.value = value;
        saveSeatingFlag(birthInput.dataset.rosterBirth, { birthYear: value }, true);
        return;
      }

      // --- 명부: 객실 유형 ---
      const roomTypeSelect = event.target.closest("[data-roster-roomtype]");
      if (roomTypeSelect) {
        saveSeatingFlag(roomTypeSelect.dataset.rosterRoomtype,
          { roomType: roomTypeSelect.value || "double" }, true);
        return;
      }

      // --- 자리 배치: 고정 테이블 ---
      const fixedTable = event.target.closest("[data-fixed-table]");
      if (fixedTable) {
        const value = String(fixedTable.value || "").replace(/[^\d]/g, "").slice(0, 3);
        fixedTable.value = value;
        saveSeatingFlag(fixedTable.dataset.fixedTable, { fixedTable: value }, true);
        return;
      }

      // --- 개최 공문 판독 ---
      if (event.target.id === "courseDocumentInput") {
        const file = event.target.files && event.target.files[0];
        const status = document.getElementById("courseDocumentStatus");
        if (status && file) status.textContent = file.name + " · " + formatBytes(file.size);
        event.target.value = "";
        if (file) parseCourseDocument(file);
        return;
      }

      // --- 명부 엑셀 업로드 ---
      if (event.target.id === "rosterFileInput") {
        const file = event.target.files && event.target.files[0];
        event.target.value = "";
        if (file) handleRosterFile(file);
        return;
      }

      // --- 이미 만든 배치표(좌석·방·식사 통합) ---
      if (event.target.id === "preparedPlanInput") {
        const file = event.target.files && event.target.files[0];
        event.target.value = "";
        if (file) handlePreparedPlanFile(file);
        return;
      }

      // --- 수동 배치표 업로드 ---
      if (event.target.id === "manualSeatingInput") {
        const file = event.target.files && event.target.files[0];
        event.target.value = "";
        if (file) handleManualSeatingFile(file);
        return;
      }
      if (event.target.matches("[data-select-all-courses]")) {
        STATE.selectedCourseIds = event.target.checked
          ? (STATE.courses || []).map(function (course) { return course.siteId; })
          : [];
        renderCourseList("manage");
        return;
      }
      if (event.target.matches("[data-select-all-trash]")) {
        STATE.selectedTrashIds = event.target.checked
          ? (STATE.trashCourses || []).map(function (course) { return course.siteId; })
          : [];
        renderTrash();
        return;
      }
      if (event.target.name === "satisfactionType") {
        STATE.satisfactionType = event.target.value;
        refreshSatisfactionCreateTab();
        return;
      }
      if (event.target.id === "surveyProgramFilter") {
        STATE.satisfactionFilter.programName = event.target.value;
        loadSatisfactionList(false);
        return;
      }
      if (event.target.id === "courseMealSkip") {
        const root = document.getElementById("courseMealSlots");
        if (root) {
          root.classList.toggle("meal-slot-disabled", event.target.checked);
          root.classList.remove("meal-slot-invalid");
        }
        updateMealSelectionSummary();
        updateCreateSummary();
      }
      if (event.target.id === "courseDays") {
        renderCreateDateAndMealFields();
        updateCreateSummary();
      }
      // 달력에서 고른 경우 → 년·월·일 칸에 반영
      const pickedDate = /^courseDate(\d+)$/.exec(event.target.id);
      if (pickedDate) {
        const day = Number(pickedDate[1]);
        syncHiddenToKoreanDate(day);
        if (day === 1) autofillCourseDatesFromFirst();
        else { syncCreatePeriodDefaults(false); refreshMealTileLabels(); }
        setFieldError("courseDate" + day, "");
      }
      // 년·월·일 칸을 직접 고친 경우 → 숨은 date 입력에 반영
      const typedDate = /^courseDate[YMD](\d+)$/.exec(event.target.id);
      if (typedDate) {
        const day = Number(typedDate[1]);
        syncKoreanDateToHidden(day);
        if (day === 1) autofillCourseDatesFromFirst();
        else { syncCreatePeriodDefaults(false); refreshMealTileLabels(); }
        setFieldError("courseDate" + day, "");
      }
      if (event.target.name === "courseType") {
        toggleCreateCourseType(event.target.value);
        updateCreateSummary();
      }
      if (event.target.dataset.uploadType) handleUploadSelection(event.target.dataset.uploadType, event.target.files);
      if (event.target.dataset.mealFile) handleMealFile(event.target.dataset.mealFile, event.target.files[0]);

    function formatPhoneInput(input) {
      const digits = String(input.value || "").replace(/\D/g, "").slice(0, 11);
      let formatted = digits;
      if (digits.indexOf("02") === 0) {
        formatted = digits.length <= 2 ? digits : digits.length <= 5
          ? digits.slice(0, 2) + "-" + digits.slice(2)
          : digits.length <= 9
            ? digits.slice(0, 2) + "-" + digits.slice(2, 5) + "-" + digits.slice(5)
            : digits.slice(0, 2) + "-" + digits.slice(2, 6) + "-" + digits.slice(6);
      } else if (digits.length > 3) {
        formatted = digits.length <= 7
          ? digits.slice(0, 3) + "-" + digits.slice(3)
          : digits.slice(0, 3) + "-" + digits.slice(3, digits.length - 4) + "-" + digits.slice(-4);
      }
      if (input.value !== formatted) input.value = formatted;
    }

    function validateProfileField(id, liveMode) {
      const element = document.getElementById(id);
      if (!element || !/^profile/.test(id)) return true;
      const value = String(element.value || "").trim();
      let message = "";
      const phonePattern = /^\d{2,4}-\d{3,4}-\d{4}$/;

      if (id === "profileName" && !value) message = "이름을 입력해주세요.";
      if (id === "profileOrganization" && !value) message = "소속 기관명을 입력해주세요.";
      if (id === "profilePosition" && !value) message = "직책을 입력해주세요.";
      if (id === "profileLandline") {
        if (!value) message = "유선번호를 입력해주세요.";
        else if (!phonePattern.test(value)) {
          message = "하이픈(-)을 포함해 입력해주세요. 예: 02-3017-0000";
        }
      }
      if (id === "profileMobile" && value && !phonePattern.test(value)) {
        message = "하이픈(-)을 포함해 입력해주세요. 예: 010-1234-5678";
      }
      if (id === "profileEmail") {
        if (!value) message = "이메일을 입력해주세요.";
        else if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(value)) {
          message = "@를 포함한 올바른 이메일을 입력해주세요.";
        }
      }

      const errorElement = document.getElementById(id + "Error");
      element.classList.toggle("field-invalid", Boolean(message));
      element.setAttribute("aria-invalid", message ? "true" : "false");
      if (errorElement) {
        errorElement.textContent = message;
        errorElement.classList.toggle("hidden", !message);
      }
      return !message;
    }

    function validateProfileForm() {

    }

    function checkPasswordPolicy(value) {
      const password = String(value || "");
      if (password.length < PASSWORD_POLICY.minLength) {
        return "비밀번호는 " + PASSWORD_POLICY.minLength + "자 이상이어야 합니다.";
      }
      if (password.length > PASSWORD_POLICY.maxLength) {
        return "비밀번호는 " + PASSWORD_POLICY.maxLength + "자 이하로 입력해주세요.";
      }
      if (/\s/.test(password)) return "비밀번호에는 공백을 사용할 수 없습니다.";
      if (!/[A-Za-z]/.test(password)) return "비밀번호에 영문을 최소 1자 포함해주세요.";
      if (!/\d/.test(password)) return "비밀번호에 숫자를 최소 1자 포함해주세요.";
      return "";
    }

    /** 새 비밀번호 / 확인란을 함께 검증합니다. 통과하면 빈 문자열을 돌려줍니다. */
    function validateNewPasswordPair(newId, confirmId, currentId) {
      const newPassword = valueOf(newId);
      const confirmPassword = valueOf(confirmId);
      const policyMessage = checkPasswordPolicy(newPassword);
      if (policyMessage) return policyMessage;
      if (newPassword !== confirmPassword) return "새 비밀번호 두 값이 일치하지 않습니다.";
      if (currentId && newPassword === valueOf(currentId)) {
        return "현재 비밀번호와 다른 비밀번호를 입력해주세요.";
      }
      return "";
    }

    function setModalBusy(formId, busy, busyLabel) {
      const form = document.getElementById(formId);
      if (!form) return;
      const button = form.querySelector('button[type="submit"]');
      if (!button) return;
      if (busy) {
        if (!button.dataset.idleLabel) button.dataset.idleLabel = button.innerHTML;
        button.disabled = true;
        button.innerHTML = '<span class="btn-spinner" aria-hidden="true"></span>' +
          '<span>' + escapeHtml(busyLabel || "처리 중입니다...") + '</span>';
      } else {
        button.disabled = false;
        if (button.dataset.idleLabel) button.innerHTML = button.dataset.idleLabel;
      }
      form.classList.toggle("modal-form-busy", Boolean(busy));
    }

    /** '다음에 변경하기' — 이번 접속에서만 건너뜁니다. */
    function deferPasswordChange() {
      setModalBusy("passwordChangeForm", true, "잠시만 기다려주세요...");
      callServer("portalDeferPasswordChange", STATE.token)
        .then(function (result) {
          ELEMENTS.modalRoot.dataset.required = "false";
          closeModal(true);
          showToast("비밀번호 변경을 미뤘습니다.", result && result.message, "info");
          return navigate("dashboard");
        })
        .catch(function (error) {
          setModalBusy("passwordChangeForm", false);
          showHelpResult("passwordChangeResult", "error",
            String((error && error.message) || "잠시 후 다시 시도해주세요."));
        });
    }

    function handleModalClick(event) {
      const operationRemove = event.target.closest("[data-confirm-operation-remove]");
      if (operationRemove) {
        const taskId = operationRemove.dataset.confirmOperationRemove;
        const plan = ensureOperationsPlan();
        plan.tasks = (plan.tasks || []).filter(function (task) {
          return task.id !== taskId || task.system;
        });
        closeModal(true);
        persistOperationsPlan("추가 업무를 삭제했습니다.");
        return;
      }
      const discardPageBuilder = event.target.closest("[data-discard-page-builder]");
      if (discardPageBuilder) {
        const target = discardPageBuilder.dataset.discardPageBuilder || "dashboard";
        STATE.pageBuilderDirty = false;
        closeModal(true);
        navigate(target, { skipPageBuilderGuard: true });
        return;
      }
      const pageBlockRemove = event.target.closest("[data-confirm-page-block-remove]");
      if (pageBlockRemove) {
        const index = Number(pageBlockRemove.dataset.confirmPageBlockRemove);
        if (index >= 0 && index < STATE.pageBlocks.length) {
          STATE.pageBlocks.splice(index, 1);
          STATE.pageBuilderDirty = true;
        }
        closeModal(true);
        updateCourseTabContent();
        showToast("카드를 삭제했습니다.", "실제 사용자 사이트에 반영하려면 변경사항을 저장해주세요.");
        return;
      }
      // 공문 판독 결과 적용
      if (event.target.closest('[data-action="apply-document-draft"]')) {
        applyDocumentDraft();
        return;
      }
      // 공개 시각 일괄 채우기 적용
      if (event.target.closest('[data-action="apply-schedule-plan"]')) {
        applySchedulePlan();
        return;
      }
      // 교육 복제 실행
      if (event.target.closest('[data-action="run-duplicate-course"]')) {
        runDuplicateCourse();
        return;
      }
      // 개선 제안
      if (event.target.closest('[data-action="apply-suggestion"]')) {
        applySeatingSuggestion();
        return;
      }
      if (event.target.closest('[data-action="ignore-suggestion"]')) {
        const box = event.target.closest(".suggest-box");
        if (box) box.innerHTML = '<p class="suggest-kept">알겠습니다. 올려주신 배치 그대로 확정하시면 됩니다.</p>';
        return;
      }
      const tempPasswordCopy = event.target.closest("[data-copy-temp-password]");
      if (tempPasswordCopy) {
        copyText(tempPasswordCopy.dataset.copyTempPassword, "임시 비밀번호를 복사했습니다.");
        return;
      }
      // 엑셀 명단 반영
      if (event.target.closest('[data-action="apply-roster-rows"]')) {
        applyRosterRows();
        return;
      }
      // 이미 만든 배치표 반영
      if (event.target.closest('[data-action="apply-prepared-plan"]')) {
        applyPreparedPlan();
        return;
      }
      // 성별이 다른 룸메이트 요청 처리
      if (event.target.closest('[data-action="room-force-mixed"]')) {
        closeModal();
        generateRoomAssignment(true);
        return;
      }
      if (event.target.closest('[data-action="room-force-separate"]')) {
        closeModal();
        STATE.roomIgnoreMixed = true;
        generateRoomAssignmentIgnoringMixed();
        return;
      }

      // 초기 비밀번호 변경 건너뛰기
      if (event.target.closest('[data-action="defer-password-change"]')) {
        deferPasswordChange();
        return;
      }

      // 비밀번호 마스킹 해제
      const peek = event.target.closest("[data-password-peek]");
      if (peek) {
        const field = document.getElementById(peek.dataset.passwordPeek);
        if (field) {
          const shown = field.type === "text";
          field.type = shown ? "password" : "text";
          peek.textContent = shown ? "보기" : "숨김";
          peek.setAttribute("aria-label", shown ? "비밀번호 표시" : "비밀번호 숨기기");
          peek.classList.toggle("active", !shown);
        }
        return;
      }

      // 팝업은 '닫기' 버튼과 ESC로만 닫습니다.
      // 예전에는 바깥 여백을 클릭해도 닫혀, 작성 중이던 내용이 사라지곤 했습니다.
      if (event.target.closest("[data-modal-close]")) {
        if (ELEMENTS.modalRoot.dataset.required === "true") return;
        closeModal();
        return;
      }
      if (event.target === ELEMENTS.modalRoot) return;
      const roomSaveConfirm = event.target.closest("[data-confirm-room-save]");
      if (roomSaveConfirm) {
        closeModal();
        saveRoomAssignments(true);
        return;
      }

      const folderStep = event.target.closest("[data-survey-folder]");
      if (folderStep) {
        openSurveyFolderPicker(folderStep.dataset.surveyFolder);
        return;
      }

      const folderChoose = event.target.closest('[data-action="survey-choose-folder"]');
      if (folderChoose) {
        closeModal();
        showToast("저장 위치를 선택했습니다.",
          (STATE.satisfactionFolder && STATE.satisfactionFolder.currentPath) || "");
        refreshSatisfactionCreateTab();
        return;
      }

      const surveyRemove = event.target.closest("[data-confirm-survey-remove]");
      if (surveyRemove) {
        const surveyId = surveyRemove.dataset.confirmSurveyRemove;
        closeModal();
        showLoader("목록에서 제거 중", "설문 기록을 정리하고 있습니다.");
        api("portalDeleteSatisfactionSurvey", surveyId)
          .then(function (result) {
            STATE.satisfactionList = null;
            STATE.satisfactionInstructors = null;
            showToast("제거했습니다.", result.message);
            return loadSatisfactionList(true);
          })
          .catch(showErrorToast)
          .finally(hideLoader);
        return;
      }

      const gotoStatus = event.target.closest('[data-action="survey-goto-status"]');
      if (gotoStatus) {
        closeModal();
        STATE.satisfactionTab = "status";
        STATE.satisfactionList = null;
        renderSatisfaction();
        return;
      }

      const adminActionConfirm = event.target.closest("[data-confirm-admin-action]");
      if (adminActionConfirm) {
        const reset = adminActionConfirm.dataset.confirmAdminAction === "reset";
        const targetId = adminActionConfirm.dataset.adminId;
        closeModal();
        showLoader(
          reset ? "비밀번호 초기화 중" : "접속 제한 해제 중",
          "관리자 계정 상태를 갱신하고 있습니다."
        );
        api(reset ? "portalResetAdminPassword" : "portalUnlockAdminAccount", targetId)
          .then(function (result) {
            if (reset) {
              const temporaryPassword = String(result.temporaryPassword || "");
              showModal(
                "임시 비밀번호가 발급되었습니다",
                "이 창을 닫으면 관리자 화면에서 다시 확인할 수 없습니다.",
                '<div class="temporary-password-box"><span>대상 계정 ' + escapeHtml(result.adminId || targetId) + '</span>' +
                  '<code>' + escapeHtml(temporaryPassword) + '</code>' +
                  '<small>대상자에게 직접 전달하세요. 첫 로그인 뒤 새 비밀번호 설정이 강제됩니다.</small></div>',
                '<button type="button" class="btn btn-secondary" data-modal-close>확인 후 닫기</button>' +
                  '<button type="button" class="btn btn-primary" data-copy-temp-password="' +
                    escapeAttr(temporaryPassword) + '">임시 비밀번호 복사</button>'
              );
            } else {
              showToast("제한을 해제했습니다.", result.message);
            }
            return loadInquiries(true);
          })
          .catch(showErrorToast)
          .finally(hideLoader);
        return;
      }

      const courseTrashConfirm = event.target.closest("[data-confirm-course-trash]");
      if (courseTrashConfirm) {
        const siteIds = STATE.selectedCourseIds.slice();
        closeModal();
        showLoader("강의를 휴지통으로 이동 중", "사용자 사이트 접근을 중단하고 기록을 정리하고 있습니다.");
        api("portalMoveCoursesToTrash", siteIds)
          .then(function (result) {
            STATE.lastTrashedIds = siteIds.slice();
            STATE.selectedCourseIds = [];
            STATE.courseDeleteMode = false;
            STATE.courseListCache = {};
            STATE.dashboardFetchedAt = 0;
            showToast("휴지통으로 이동했습니다.", result.message);
            showModal(
              "휴지통으로 이동했습니다",
              "실수로 삭제했다면 지금 바로 원래 상태로 되돌릴 수 있습니다.",

    function handleModalChange(event) {
      if (event.target.id === "pageBlockType" || event.target.id === "pageBlockScheduled") {
        updatePageBlockModalFields();
        return;
      }
      if (event.target.id === "pageBlockFile") {
        const file = event.target.files && event.target.files[0];
        STATE.pendingPageBlockFile = file || null;
        const status = document.getElementById("pageBlockFileStatus");
        if (status) status.textContent = file ? file.name + " · " + formatBytes(file.size) : "PDF·PPT·PPTX · 최대 8MB";
        return;
      }
      if (event.target.id !== "editCourseDays") return;
      const days = Number(event.target.value || 1);
      const dateRoot = document.getElementById("editDateFields");
      const mealRoot = document.getElementById("editMealSlots");
      const savedDates = {};
      const savedMeals = {};
      if (dateRoot) {
        dateRoot.querySelectorAll("input").forEach(function (input) {
          savedDates[input.id] = input.value;
        });
      }
      if (mealRoot) {
        mealRoot.querySelectorAll(".meal-checkbox").forEach(function (input) {
          savedMeals[input.value] = input.checked;
        });
      }
      if (dateRoot) {
        dateRoot.innerHTML = Array.from({ length: days }, function (_, index) {
          const day = index + 1;
          const existing = savedDates["editDate" + day] || toDateInput(STATE.course["day" + day + "Date"]) || defaultCourseDate(index);
          return '<div class="form-field"><label for="editDate' + day + '">' + day + '일차</label><input id="editDate' + day + '" type="date" value="' + escapeAttr(existing) + '" required></div>';
        }).join("");
      }
      if (mealRoot) {
        mealRoot.innerHTML = Array.from({ length: days }, function (_, index) {
          const day = index + 1;
          const breakfastKey = day + "|breakfast";
          const lunchKey = day + "|lunch";
          const dinnerKey = day + "|dinner";
          const existingBreakfast = savedMeals[breakfastKey] != null
            ? savedMeals[breakfastKey]
            : (STATE.course.mealSlots || []).some(function (slot) { return Number(slot.day) === day && slot.type === "breakfast"; });
          const existingLunch = savedMeals[lunchKey] != null
            ? savedMeals[lunchKey]
            : (STATE.course.mealSlots || []).some(function (slot) { return Number(slot.day) === day && slot.type === "lunch"; });
          const existingDinner = savedMeals[dinnerKey] != null
            ? savedMeals[dinnerKey]
            : (STATE.course.mealSlots || []).some(function (slot) { return Number(slot.day) === day && slot.type === "dinner"; });
          return '<label class="check-tile"><input class="meal-checkbox" type="checkbox" value="' + breakfastKey + '"' + (existingBreakfast ? " checked" : "") + '><span>' + day + '일차 아침</span></label>' +
            '<label class="check-tile"><input class="meal-checkbox" type="checkbox" value="' + lunchKey + '"' + (existingLunch ? " checked" : "") + '><span>' + day + '일차 점심</span></label>' +
            '<label class="check-tile"><input class="meal-checkbox" type="checkbox" value="' + dinnerKey + '"' + (existingDinner ? " checked" : "") + '><span>' + day + '일차 저녁</span></label>';
        }).join("");
      }
    }

    function handleModalSubmit(event) {
      if (event.target.id === "operationTaskForm") {
        event.preventDefault();
        submitOperationTaskForm(event.target);
        return;
      }
      if (event.target.id === "operationsFunnelForm") {
        event.preventDefault();
        submitOperationsFunnelForm();
        return;
      }
      if (event.target.id === "userPageBlockForm") {
        event.preventDefault();
        submitUserPageBlockForm(event.target);
        return;
      }
      if (event.target.id === "findIdForm") {
        event.preventDefault();
        const name = valueOf("findIdName");
        const email = valueOf("findIdEmail");
        if (!name || !email) {

          "newPasswordConfirm",
          "currentPassword"

        if (invalidMessage) {
          showHelpResult("passwordChangeResult", "error", invalidMessage);
          return;
        }
        clearHelpResult("passwordChangeResult");
        const currentPassword = valueOf("currentPassword");

            showHelpResult("passwordChangeResult", "error",
              String((error && error.message) || "비밀번호를 변경하지 못했습니다."));

    function showModal(title, description, body, actions, wide, extraClass) {
      STATE.modalReturnFocus = document.activeElement;
      ELEMENTS.modalRoot.dataset.required = "false";
      ELEMENTS.modalRoot.innerHTML =
        '<div class="modal-card ' + (wide ? "modal-wide " : "") + (extraClass || "") + '" role="dialog" aria-modal="true" aria-labelledby="modalTitle">' +

      const first = ELEMENTS.modalRoot.querySelector("input:not([type=hidden]), select, textarea, button, a[href]");

    function closeModal(force) {
      if (!ELEMENTS.modalRoot) return;
      if (!force && ELEMENTS.modalRoot.dataset.required === "true") return;
      ELEMENTS.modalRoot.classList.add("hidden");
      ELEMENTS.modalRoot.innerHTML = "";
      ELEMENTS.modalRoot.dataset.required = "false";
      document.body.style.overflow = "";
      if (STATE.modalReturnFocus && document.contains(STATE.modalReturnFocus)) {
        try { STATE.modalReturnFocus.focus(); } catch (error) {}
      }
      STATE.modalReturnFocus = null;
    }

    function trapModalFocus(event) {
      const focusable = Array.from(ELEMENTS.modalRoot.querySelectorAll('a[href],button:not([disabled]),input:not([disabled]):not([type="hidden"]),select:not([disabled]),textarea:not([disabled]),[tabindex]:not([tabindex="-1"])'));
      if (!focusable.length) return;
      const first = focusable[0];
      const last = focusable[focusable.length - 1];
      if (event.shiftKey && document.activeElement === first) { event.preventDefault(); last.focus(); }
      else if (!event.shiftKey && document.activeElement === last) { event.preventDefault(); first.focus(); }
    }

    /** 입력칸을 빨갛게 하고 라벨 옆에 짧은 경고를 답니다. message가 비면 해제합니다. */
    function setFieldError(id, message) {
      const input = document.getElementById(id);
      const label = document.getElementById(id + "Error");
      if (input) {
        input.classList.toggle("field-invalid", Boolean(message));
        input.setAttribute("aria-invalid", message ? "true" : "false");
      }
      if (label) {
        label.textContent = message || "";
        label.classList.toggle("hidden", !message);
      } else if (message) {
        // 숨김 입력처럼 경고를 붙일 자리가 없으면 최소한 토스트로라도 알립니다.
        showToast("입력값을 확인해주세요.", message, "error");
      }
      return !message;
    }

    function copyText(text, successMessage) {
      const value = String(text || "");
      if (!value) {
        showToast("복사할 주소가 없습니다.", "", "error");
        return;
      }
      const fallback = function () {
        const textarea = document.createElement("textarea");
        textarea.value = value;
        textarea.style.position = "fixed";
        textarea.style.opacity = "0";
        document.body.appendChild(textarea);
        textarea.select();
        document.execCommand("copy");
        textarea.remove();
      };
      if (navigator.clipboard && navigator.clipboard.writeText) {
        navigator.clipboard.writeText(value).catch(fallback);
      } else {
        fallback();
      }
      showToast("복사했습니다.", successMessage || "");
    }

    function readKoreanDate(value) {
      const match = String(value || "").match(/(\d{4})\D+(\d{1,2})\D+(\d{1,2})/);
      if (!match) return null;
      return new Date(Number(match[1]), Number(match[2]) - 1, Number(match[3]));
    }

    function toDateInput(value) {
      const date = readKoreanDate(value);
      if (!date) return "";
      return date.getFullYear() + "-" + pad(date.getMonth() + 1) + "-" + pad(date.getDate());
    }

    function shortCourseDate(course) {
      const date = readKoreanDate(course.day1Date);
      return date ? (date.getMonth() + 1) + "월\n" + date.getDate() + "일" : (course.generation || "교육").slice(0, 5);
    }

    function defaultCourseDate(offset) {
      const date = new Date();
      date.setDate(date.getDate() + 7 + Number(offset || 0));
      return date.getFullYear() + "-" + pad(date.getMonth() + 1) + "-" + pad(date.getDate());
    }

    function toDateTimeLocal(value) {
      if (!value) return "";
      const date = new Date(value);
      if (isNaN(date.getTime())) return String(value).slice(0, 16);
      return toLocalDateTimeString(date);
    }

    function toLocalDateTimeString(date) {
      return date.getFullYear() + "-" + pad(date.getMonth() + 1) + "-" + pad(date.getDate()) + "T" + pad(date.getHours()) + ":" + pad(date.getMinutes());
    }

    function localInputToIso(value) {
      if (!value) return "";
      const date = new Date(value);
      return isNaN(date.getTime()) ? "" : date.toISOString();
    }

    function startServerClock() {
      if (STATE.serverClockTimer) {

      if (STATE.serverClockSyncTimer) {
        window.clearTimeout(STATE.serverClockSyncTimer);
      }

      function renderClock() {
        const serverDate = new Date(Date.now() + STATE.serverClockOffset);
        ELEMENTS.topbarDate.textContent = formatServerClock(serverDate);
      }

      renderClock();

      STATE.serverClockSyncTimer = window.setTimeout(function () {
        if (!STATE.token) return;

          .then(function (result) {

              Number(result.epochMs || Date.now()) - Date.now();

          })
          .catch(function () {
            STATE.serverClockOffset = 0;
          });
      }, 1400);

      STATE.serverClockTimer = window.setInterval(renderClock, 1000);
      startNetworkMonitor();
    }

    function formatServerClock(date) {
      const formatter = new Intl.DateTimeFormat("ko-KR", {
        timeZone: "Asia/Seoul",
        year: "numeric",
        month: "2-digit",
        day: "2-digit",
        hour: "2-digit",
        minute: "2-digit",
        hourCycle: "h23"
      });

      const parts = {};
      formatter.formatToParts(date).forEach(function (part) {
        if (part.type !== "literal") parts[part.type] = part.value;
      });

      return (
        parts.year + "년 " +
        parts.month + "월 " +
        parts.day + "일 " +
        parts.hour + ":" +
        parts.minute
      );
    }

      return "GOOD EVENING";
    }

      return "portal_" + Date.now().toString(36) + "_" + Math.random().toString(36).slice(2, 10);
    }

    function guessMimeType(fileName) {
      const extension = String(fileName || "").toLowerCase().split(".").pop();
      const map = {
        pdf: "application/pdf",
        ppt: "application/vnd.ms-powerpoint",
        pptx: "application/vnd.openxmlformats-officedocument.presentationml.presentation",
        jpg: "image/jpeg",
        jpeg: "image/jpeg",
        png: "image/png",
        gif: "image/gif",
        webp: "image/webp",
        heic: "image/heic",
        heif: "image/heif"
      };
      return map[extension] || "application/octet-stream";
    }

    function formatBytes(bytes) {
      const value = Number(bytes || 0);
      if (value < 1024) return value + " B";
      if (value < 1024 * 1024) return (value / 1024).toFixed(1) + " KB";
      return (value / 1024 / 1024).toFixed(1) + " MB";
    }

    function clamp(value, min, max) {
      return Math.max(min, Math.min(max, Number(value || 0)));
    }

    function pad(value) {
      return String(value).padStart(2, "0");
    }

    function escapeHtml(value) {
      return String(value == null ? "" : value)
        .replace(/&/g, "&amp;")
        .replace(/</g, "&lt;")
        .replace(/>/g, "&gt;")
        .replace(/"/g, "&quot;")
        .replace(/'/g, "&#39;");
    }

    function escapeAttr(value) {
      return escapeHtml(value).replace(/`/g, "&#96;");
    }
