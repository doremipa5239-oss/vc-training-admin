    function satisfactionTabButton(key, label) {
      const active = STATE.satisfactionTab === key;
      return '<button type="button" class="course-tab ' + (active ? "active" : "") +
        '" data-satisfaction-tab="' + key + '" role="tab" aria-selected="' + (active ? "true" : "false") +
        '">' + escapeHtml(label) + '</button>';
    }

    function renderSatisfactionTab(navigationVersion) {
      const root = document.getElementById("satisfactionTabContent");
      if (!root) return Promise.resolve();
      document.querySelectorAll("[data-satisfaction-tab]").forEach(function (button) {
        const active = button.dataset.satisfactionTab === STATE.satisfactionTab;
        button.classList.toggle("active", active);
        button.setAttribute("aria-selected", active ? "true" : "false");
      });
      if (STATE.satisfactionTab === "status") return loadSatisfactionList(false, navigationVersion);
      if (STATE.satisfactionTab === "analysis") return renderSatisfactionAnalysisTab(navigationVersion);
      root.innerHTML = renderSatisfactionCreateTab();
      updateSurveyWizard();
      return Promise.resolve();
    }

    function updateSurveyWizard() {
      const step = clamp(Number(STATE.satisfactionStep || 1), 1, 3);
      STATE.satisfactionStep = step;
      document.querySelectorAll("[data-survey-step]").forEach(function (section) {
        section.classList.toggle("active", Number(section.dataset.surveyStep) === step);
      });
      document.querySelectorAll("[data-survey-progress]").forEach(function (item) {
        const itemStep = Number(item.dataset.surveyProgress);
        item.classList.toggle("active", itemStep === step);
        item.classList.toggle("completed", itemStep < step);
      });
      const previous = document.getElementById("surveyPrevButton");
      const next = document.getElementById("surveyNextButton");
      const submit = document.getElementById("surveySubmitButton");
      const hint = document.getElementById("surveyStepHint");
      if (previous) previous.classList.toggle("hidden", step === 1);
      if (next) next.classList.toggle("hidden", step === 3);
      if (submit) submit.classList.toggle("hidden", step !== 3);
      if (hint) hint.textContent = step + " / 3 단계";
    }

    function moveSurveyWizard(direction) {
      STATE.satisfactionStep = clamp(Number(STATE.satisfactionStep || 1) + direction, 1, 3);
      updateSurveyWizard();
      // 단계 전환 시 자동 스크롤은 화면 흔들림만 유발해 제거했습니다.
    }

    function addSatisfactionLecture() {
      const lecture = {
        lecturePeriod: valueOf("surveyLecturePeriod"),
        lectureTitle: valueOf("surveyLectureTitle"),
        instructorName: valueOf("surveyInstructorName"),
        instructorAffiliation: valueOf("surveyInstructorAffiliation")
      };
      if (!lecture.lecturePeriod || !lecture.lectureTitle || !lecture.instructorName) {
        showToast("강의 정보를 확인해주세요.", "차수, 강의명, 강사명은 반드시 입력해야 합니다.", "error");
        return;
      }
      STATE.satisfactionLectures.push(lecture);
      refreshSatisfactionCreateTab();
      const focusTarget = document.getElementById("surveyLecturePeriod");
      if (focusTarget) focusTarget.focus();
    }

    function submitSatisfactionSurvey() {
      const overall = STATE.satisfactionType === "overall";
      const programName = valueOf("surveyProgramName");
      if (!programName) {
        showToast("과정명을 입력해주세요.", "설문지 제목에 사용됩니다.", "error");
        return;
      }
      const payload = {
        formType: overall ? "overall" : "lecture",
        programName: programName,
        siteId: valueOf("surveyLinkedCourse"),
        targetFolderId: STATE.satisfactionFolder ? STATE.satisfactionFolder.currentId : ""
      };

      if (!overall) {
        if (STATE.satisfactionLectures.length) {
          payload.lectures = STATE.satisfactionLectures.slice();
        } else {
          payload.lecturePeriod = valueOf("surveyLecturePeriod");
          payload.lectureTitle = valueOf("surveyLectureTitle");
          payload.instructorName = valueOf("surveyInstructorName");
          payload.instructorAffiliation = valueOf("surveyInstructorAffiliation");
          if (!payload.lecturePeriod || !payload.lectureTitle || !payload.instructorName) {
            showToast("강의 정보를 확인해주세요.", "차수, 강의명, 강사명을 입력하거나 목록에 추가해주세요.", "error");
            return;
          }
        }
      }

      const count = payload.lectures ? payload.lectures.length : 1;
      showLoader("설문지 생성 중", count + "개 설문지를 만들고 있습니다. 잠시만 기다려주세요.");
      api("portalCreateSatisfactionSurvey", payload)
        .then(function (result) {
          STATE.satisfactionLectures = [];
          STATE.satisfactionList = null;
          STATE.satisfactionInstructors = null;
          STATE.satisfactionStep = 1;
          showSatisfactionResult(result);
        })
        .catch(showErrorToast)
        .finally(hideLoader);
    }

    /* 응답 수는 구글 폼을 하나씩 열어야 알 수 있어 설문 수만큼 시간이 걸립니다.
       그래서 두 단계로 나눕니다.
         1단계 목록만 즉시 표시(응답 수는 '확인 중')
         2단계 응답 수를 채워 다시 그리기
       사용자는 표를 바로 보게 되고, 숫자만 잠시 뒤에 채워집니다. */
    function loadSatisfactionList(forceRefresh, navigationVersion) {
      const requestVersion = Number(navigationVersion || STATE.navigationVersion);
      const root = document.getElementById("satisfactionTabContent");
      if (!root) return Promise.resolve();
      if (STATE.satisfactionList && !forceRefresh) {
        root.innerHTML = renderSatisfactionStatusTab();
        return Promise.resolve();
      }

      const filter = {
        programName: STATE.satisfactionFilter.programName,
        formType: STATE.satisfactionFilter.formType
      };
      const stillHere = function () {
        return requestVersion === STATE.navigationVersion &&
          STATE.view === "satisfaction" && STATE.satisfactionTab === "status";
      };
      const paint = function () {
        const target = document.getElementById("satisfactionTabContent");
        if (target) target.innerHTML = renderSatisfactionStatusTab();
      };

      root.innerHTML = renderTabLoading("설문 목록을 불러오는 중입니다.");

      return api("portalListSatisfactionSurveys", Object.assign({ withCounts: false }, filter))
        .then(function (listOnly) {
          if (!stillHere()) return listOnly;
          STATE.satisfactionList = listOnly;
          STATE.satisfactionPrograms = listOnly.programs || [];
          STATE.satisfactionCountsLoading = Boolean((listOnly.items || []).length);
          paint();
          if (!STATE.satisfactionCountsLoading) return listOnly;

          return api("portalListSatisfactionSurveys",
            Object.assign({ withCounts: true, forceRefresh: Boolean(forceRefresh) }, filter))
            .then(function (withCounts) {
              STATE.satisfactionCountsLoading = false;
              if (!stillHere()) return withCounts;
              STATE.satisfactionList = withCounts;
              paint();
              return withCounts;
            })
            .catch(function (error) {
              STATE.satisfactionCountsLoading = false;
              if (stillHere()) paint();
              showToast("응답 수를 불러오지 못했습니다.", error.message, "warning");
            });
        })
        .catch(function (error) {
          STATE.satisfactionCountsLoading = false;
          const target = document.getElementById("satisfactionTabContent");
          if (target) {
            target.innerHTML = '<div class="panel">' +
              renderEmpty("!", "참여 현황을 불러오지 못했습니다.", error.message, "", "") + '</div>';
          }
        });
    }

                '<th>과정</th><th>구분</th><th>설문 제목</th><th>응답</th><th>최근 응답</th><th></th>' +

    }

    function satisfactionTypeFilter(value, label) {

        '" data-survey-type-filter="' + escapeAttr(value) + '">' + escapeHtml(label) + '</button>';
    }

    function renderSatisfactionStatusRow(item) {
      const lecture = item.formType === "lecture";
      const title = lecture
        ? (item.lecturePeriod ? "[" + item.lecturePeriod + "] " : "") + item.lectureTitle +
          (item.instructorName ? " · " + item.instructorName : "")
        : "과정 전체 만족도 조사";
      return '<tr>' +
        '<td>' + escapeHtml(item.programName) + '</td>' +
        '<td><span class="status-chip ' + (lecture ? "status-upcoming" : "status-active") + '">' +
          (lecture ? "강의 평가" : "전체 만족도") + '</span></td>' +
        '<td class="survey-cell-title">' + escapeHtml(title) + '</td>' +
        '<td><strong class="survey-count">' + (
          item.readError ? "확인 불가"
            : item.responseCount == null ? '<span class="survey-count-pending">확인 중…</span>'

        '<td>' + escapeHtml(item.lastResponseAt || "-") + '</td>' +

          '<a class="btn btn-soft btn-small" href="' + escapeAttr(item.publicUrl) +
            '" target="_blank" rel="noopener">응답 링크</a>' +
          '<button type="button" class="btn btn-danger btn-small" data-action="survey-remove-record"' +
            ' data-survey-id="' + escapeAttr(item.surveyId) + '"' +
            ' data-survey-title="' + escapeAttr(item.programName) + '">목록 제거</button>' +
        '</td>' +
      '</tr>';
    }

    /* ---------- 탭 3. 결과 분석 ---------- */

    function renderSatisfactionAnalysisTab(navigationVersion) {
      const root = document.getElementById("satisfactionTabContent");
      if (!root) return Promise.resolve();
      const modeTabs =
        '<div class="filter-toolbar"><div class="filter-tabs">' +
          '<button type="button" class="filter-tab ' +
            (STATE.satisfactionAnalysisMode === "survey" ? "active" : "") +
            '" data-survey-analysis-mode="survey">설문별 결과</button>' +
          '<button type="button" class="filter-tab ' +
            (STATE.satisfactionAnalysisMode === "instructor" ? "active" : "") +
            '" data-survey-analysis-mode="instructor">강사별 누적</button>' +
        '</div>' +
        (STATE.satisfactionAnalysisMode === "instructor"
          ? '<div class="survey-filter-right">' +
            '<button type="button" class="btn btn-secondary btn-small" data-action="survey-refresh-instructor">새로고침</button>' +
            '</div>'
          : "") +
        '</div>';

      if (STATE.satisfactionAnalysisMode === "instructor") {
        if (!STATE.satisfactionInstructors) {
          root.innerHTML = modeTabs + renderTabLoading("강사별 누적 결과를 계산하는 중입니다. 설문이 많으면 시간이 걸릴 수 있습니다.");
          return api("portalGetInstructorSatisfaction", { forceRefresh: STATE.satisfactionForceInstructor === true })
            .then(function (result) {
              STATE.satisfactionInstructors = result;
              STATE.satisfactionForceInstructor = false;
              const target = document.getElementById("satisfactionTabContent");
              if (target) target.innerHTML = modeTabs + renderInstructorAnalysis();
            })
            .catch(function (error) {
              const target = document.getElementById("satisfactionTabContent");
              if (target) {
                target.innerHTML = modeTabs + '<div class="panel">' +
                  renderEmpty("!", "강사별 결과를 불러오지 못했습니다.", error.message, "", "") + '</div>';
              }
            });
        }
        root.innerHTML = modeTabs + renderInstructorAnalysis();
        return Promise.resolve();
      }

      if (!STATE.satisfactionAnalysis) {
        root.innerHTML = modeTabs + '<div class="panel">' +
          renderEmpty("◑", "분석할 설문을 선택해주세요.",
            "‘참여 현황 보기’ 탭에서 설문의 ‘결과 보기’를 누르면 이곳에 표시됩니다.", "", "") + '</div>';
        return Promise.resolve();
      }
      root.innerHTML = modeTabs + renderSurveyAnalysis();
      return Promise.resolve();
    }

    function loadSatisfactionAnalysis(surveyId) {
      STATE.satisfactionTab = "analysis";
      STATE.satisfactionAnalysisMode = "survey";
      showLoader("설문 결과 분석 중", "응답을 읽어 문항별로 집계하고 있습니다.");
      return api("portalGetSatisfactionAnalysis", surveyId, true)
        .then(function (result) {
          STATE.satisfactionAnalysis = result;
          renderSatisfaction();
        })
        .catch(showErrorToast)
        .finally(hideLoader);
    }

    /** 5단계 분포를 가로 막대로 그립니다. */
    function scaleBar(summary) {
      const segments = SCALE_LABELS.map(function (label, index) {
        const percent = Number((summary.percents || {})[label] || 0);
        if (!percent) return "";
        return '<span class="scale-seg scale-' + SCALE_KEYS[index] + '" style="width:' + percent + '%" ' +
          'title="' + escapeAttr(label + " " + percent + "% (" + Number((summary.counts || {})[label] || 0) + "명)") +
          '"></span>';
      }).join("");
      const legend = SCALE_LABELS.map(function (label, index) {
        const count = Number((summary.counts || {})[label] || 0);
        if (!count) return "";
        return '<span class="scale-legend-item"><i class="scale-' + SCALE_KEYS[index] + '"></i>' +
          escapeHtml(label) + ' ' + count + '명 (' + Number((summary.percents || {})[label] || 0) + '%)</span>';
      }).join("");
      return '<div class="scale-bar">' + (segments || '<span class="scale-empty"></span>') + '</div>' +
        '<div class="scale-legend">' + (legend || '<span class="scale-legend-item">응답 없음</span>') + '</div>';
    }

    function scoreBadge(average, positiveRate) {
      return '<span class="score-badge"><strong>' + Number(average || 0).toFixed(2) + '</strong>' +
        '<small>5점 만점 · 긍정 ' + Number(positiveRate || 0) + '%</small></span>';
    }

    function renderSurveyAnalysis() {
      const data = STATE.satisfactionAnalysis;
      const survey = data.survey || {};
      const questions = data.questions || [];
      const choiceQuestions = questions.filter(function (q) { return q.kind !== "text"; });
      const textQuestions = questions.filter(function (q) { return q.kind === "text"; });
      const title = survey.formType === "lecture"
        ? (survey.lecturePeriod ? "[" + survey.lecturePeriod + "] " : "") + survey.lectureTitle
        : "과정 전체 만족도 조사";

      return '<div class="page-stack">' +
        '<div class="panel panel-pad survey-summary-panel">' +
          panelHeader(escapeHtml(title),
            escapeHtml(survey.programName + (survey.instructorName ? " · " + survey.instructorName : "")),
            '<a class="btn btn-secondary btn-small" href="' + escapeAttr(survey.editUrl) +
              '" target="_blank" rel="noopener">설문지 열기</a>') +
          '<div class="survey-summary-grid">' +
            surveyMetric("응답 인원", Number(data.responseCount || 0) + "명") +
            surveyMetric("전체 평균", Number(data.overallAverage || 0).toFixed(2) + " / 5") +
            surveyMetric("최근 응답", data.lastResponseAt || "-") +
            surveyMetric("응답자", (data.respondents || []).length
              ? (data.respondents || []).slice(0, 4).join(", ") +
                ((data.respondents || []).length > 4 ? " 외 " + ((data.respondents || []).length - 4) + "명" : "")
              : "-") +
          '</div>' +
        '</div>' +

        (data.responseCount
          ? '<div class="panel panel-pad">' +
              panelHeader("문항별 만족도", "문항마다 응답 인원과 5단계 비율을 보여줍니다.", "") +
              '<div class="question-list">' + choiceQuestions.map(renderQuestionBlock).join("") + '</div>' +
            '</div>' +
            '<div class="panel panel-pad">' +
              panelHeader("서술형 답변", "응답자가 직접 작성한 의견입니다.", "") +
              textQuestions.map(function (question) {
                return '<div class="text-answer-group"><h4>' + escapeHtml(question.title) + '</h4>' +
                  ((question.answers || []).length
                    ? '<div class="text-answer-list">' + question.answers.map(function (answer) {
                        return '<blockquote class="text-answer">' + escapeHtml(answer.text) +
                          '<cite>' + escapeHtml(answer.at) + '</cite></blockquote>';
                      }).join("") + '</div>'
                    : '<p class="field-help">아직 작성된 의견이 없습니다.</p>') +
                '</div>';
              }).join("") +
            '</div>'
          : '<div class="panel">' + renderEmpty("◔", "아직 응답이 없습니다.",
              "응답이 들어오면 이곳에서 문항별 비율과 서술형 답변을 확인할 수 있습니다.", "", "") + '</div>') +
      '</div>';
    }

    function surveyMetric(label, value) {
      return '<div class="survey-metric"><span>' + escapeHtml(label) + '</span>' +
        '<strong>' + escapeHtml(String(value)) + '</strong></div>';
    }

    function renderQuestionBlock(question) {
      if (question.kind === "grid") {
        return '<article class="question-block">' +
          '<div class="question-head"><h4>' + escapeHtml(question.title) + '</h4>' +
            scoreBadge(question.average, question.positiveRate) + '</div>' +
          '<div class="question-rows">' + (question.rows || []).map(function (row) {
            return '<div class="question-row"><div class="question-row-head">' +
              '<strong>' + escapeHtml(row.label) + '</strong>' +
              '<span>' + row.total + '명 · 평균 ' + Number(row.average).toFixed(2) + '</span></div>' +
              scaleBar(row) + '</div>';
          }).join("") + '</div>' +
        '</article>';
      }
      return '<article class="question-block">' +
        '<div class="question-head"><h4>' + escapeHtml(question.title) + '</h4>' +
          scoreBadge(question.average, question.positiveRate) + '</div>' +
        '<p class="question-meta">' + question.total + '명 응답</p>' +
        scaleBar(question) +
      '</article>';
    }

    function renderInstructorAnalysis() {
      const data = STATE.satisfactionInstructors || {};
      const instructors = data.instructors || [];
      if (!instructors.length) {
        return '<div class="panel">' + renderEmpty("◑", "강사별 누적 결과가 없습니다.",
          "강의 평가 설문을 만들고 응답이 쌓이면 강사별로 집계됩니다.", "", "") + '</div>';
      }
      return '<div class="page-stack">' +
        (data.failed && data.failed.length
          ? '<div class="warning-callout"><strong>' + data.failed.length +
            '개 설문은 읽지 못했습니다.</strong><span>설문지가 삭제되었거나 접근 권한이 없을 수 있습니다.</span></div>'
          : "") +
        '<div class="panel panel-pad">' +
          panelHeader("강사별 누적 만족도", "여러 과정에 걸친 강의 평가를 강사 기준으로 합산했습니다. 평균이 높은 순입니다.", "") +
          '<div class="instructor-list">' + instructors.map(renderInstructorCard).join("") + '</div>' +
        '</div>' +
      '</div>';
    }

    function renderInstructorCard(instructor) {
      return '<article class="instructor-card">' +
        '<div class="instructor-head">' +
          '<div><h4>' + escapeHtml(instructor.instructorName) +
            (instructor.instructorAffiliation
              ? ' <small>' + escapeHtml(instructor.instructorAffiliation) + '</small>'
              : "") + '</h4>' +
            '<p class="question-meta">설문 ' + instructor.surveyCount + '개 · 응답 ' +
              instructor.responseCount + '명</p></div>' +
          scoreBadge(instructor.average, instructor.positiveRate) +
        '</div>' +
        scaleBar(instructor) +
        '<div class="instructor-rows">' + (instructor.rows || []).map(function (row) {
          return '<div class="instructor-row"><span>' + escapeHtml(row.label) + '</span>' +
            '<b>' + Number(row.average).toFixed(2) + '</b></div>';
        }).join("") + '</div>' +
        '<details class="instructor-detail"><summary>강의별 결과 ' +
          (instructor.lectures || []).length + '건 · 서술형 의견 ' +
          (instructor.comments || []).length + '건</summary>' +
          '<div class="instructor-lectures">' + (instructor.lectures || []).map(function (lecture) {
            return '<div class="instructor-lecture"><span>' +
              escapeHtml(lecture.programName + " " +
                (lecture.lecturePeriod ? "[" + lecture.lecturePeriod + "] " : "") + lecture.lectureTitle) +
              '</span><b>' + Number(lecture.average).toFixed(2) + '</b>' +
              '<small>' + lecture.responseCount + '명</small></div>';
          }).join("") + '</div>' +
          ((instructor.comments || []).length
            ? '<div class="text-answer-list" style="margin-top:14px">' +
              instructor.comments.map(function (comment) {
                return '<blockquote class="text-answer">' + escapeHtml(comment.text) +
                  '<cite>' + escapeHtml(comment.programName + " · " + comment.lectureTitle) + '</cite></blockquote>';
              }).join("") + '</div>'
            : "") +
        '</details>' +
      '</article>';
    }
  })();
