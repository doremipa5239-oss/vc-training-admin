    /**
     * 한글 날짜 입력칸.
     * <input type="date">는 표시 형식이 브라우저·OS 지역설정에 묶여 있어
     * "2026년 8월 2일"처럼 보이게 바꿀 수 없습니다. 그래서 년·월·일을
     * 각각 숫자 칸으로 두고(Tab으로 이동), 달력은 옆 버튼으로 띄웁니다.
     * 실제 값은 숨겨진 date 입력(courseDate{n})에 yyyy-MM-dd로 보관해
     * 기존 저장 로직을 그대로 씁니다.
     */
    function koreanDateField(day, value) {
      return '<div class="form-field course-date-field create-date-row">' +
        '<span class="create-date-order">' + day + '</span>' +
        '<div class="create-date-copy"><label for="courseDate' + day + '">' + day + '일차 교육일' +
          '<em id="courseDate' + day + 'Error" class="field-inline-error hidden"></em></label>' +
          '<span>' + (day === 1 ? '첫날을 바꾸면 뒤 날짜를 연속으로 제안합니다.' : '연속 일정이 아니면 이 날짜만 수정하세요.') + '</span></div>' +
        '<input id="courseDate' + day + '" name="courseDate' + day + '" type="date" value="' + escapeAttr(value) + '">' +
      '</div>';
    }

    /** 년·월·일 칸의 값을 숨은 date 입력에 반영합니다. */
    function syncKoreanDateToHidden(day) {
      const year = Number(valueOf("courseDateY" + day));
      const month = Number(valueOf("courseDateM" + day));
      const date = Number(valueOf("courseDateD" + day));
      const hidden = document.getElementById("courseDate" + day);
      if (!hidden) return "";
      if (!year || !month || !date || month > 12 || date > 31) {
        hidden.value = "";
        return "";
      }
      hidden.value = year + "-" + pad(month) + "-" + pad(date);
      return hidden.value;
    }

    /** 달력에서 고른 값을 년·월·일 칸에 반영합니다. */
    function syncHiddenToKoreanDate(day) {
      const hidden = document.getElementById("courseDate" + day);
      if (!hidden) return;
      const parts = String(hidden.value || "").match(/^(\d{4})-(\d{2})-(\d{2})$/);
      if (!parts) return;
      const y = document.getElementById("courseDateY" + day);
      const m = document.getElementById("courseDateM" + day);
      const d = document.getElementById("courseDateD" + day);
      if (y) y.value = parts[1];
      if (m) m.value = String(Number(parts[2]));
      if (d) d.value = String(Number(parts[3]));
    }

    /* ==========================================================
     * 개최 공문 자동 판독 → 1단계 입력칸 자동 채우기
     * ========================================================== */
    function parseCourseDocument(file) {
      const isPdf = /\.pdf$/i.test(file.name) || file.type === "application/pdf";
      if (!isPdf) {
        showToast("PDF만 판독할 수 있습니다.",
          "한글(hwp)·워드 파일은 “PDF로 저장” 후 올려주세요.", "error");
        return;
      }
      if (file.size > 12 * 1024 * 1024) {
        showToast("파일이 너무 큽니다.", "12MB 이하 PDF만 판독할 수 있습니다.", "error");
        return;
      }
      if (file.size < 1000) {
        showToast("파일이 비어 있는 것 같습니다.", "다시 저장한 PDF로 시도해주세요.", "error");
        return;
      }
      showLoader("공문 판독 중",
        "PDF를 문서로 변환해 글자를 읽고 있습니다. 보통 10~20초, 스캔본은 30초까지 걸립니다.");
      readFileAsBase64(file)
        .then(function (payloadFile) {
          return api("portalParseCourseDocument", {
            fileName: file.name,
            mimeType: "application/pdf",
            base64: payloadFile.base64
          });
        })
        .then(function (result) {
          STATE.courseDocumentFileName = file.name;
          showDocumentDraftModal(result.draft || {}, result.excerpt || "");
        })
        .catch(showErrorToast)
        .finally(hideLoader);
    }

    const DRAFT_FIELD_LABELS = [
      ["title", "교육명"], ["generation", "기수"], ["day1Date", "1일차"],
      ["day2Date", "2일차"], ["day3Date", "3일차"], ["day4Date", "4일차"], ["day5Date", "5일차"],
      ["startTime", "시작 시각"], ["endTime", "종료 시각"],
      ["venue", "장소"], ["managerName", "담당자"], ["managerPhone", "연락처"],
      ["audience", "교육 대상"], ["capacity", "정원"], ["fee", "교육비"], ["applyDueDate", "신청 마감"]
    ];

    function showDocumentDraftModal(draft, excerpt) {
      STATE.pendingDocumentDraft = draft;
      const found = DRAFT_FIELD_LABELS.filter(function (pair) { return draft[pair[0]]; });
      const missing = DRAFT_FIELD_LABELS.filter(function (pair) { return !draft[pair[0]]; });
      const body =
        (found.length
          ? '<div class="success-callout"><strong>' + found.length + '개 항목을 읽었습니다.</strong>' +
            '<span>확인 후 ‘이 내용으로 채우기’를 누르면 입력칸에 들어갑니다.</span></div>'
          : '<div class="warning-callout"><strong>읽어낸 항목이 없습니다.</strong>' +
            '<span>스캔 화질이 낮거나 표로만 된 공문일 수 있습니다. 직접 입력해주세요.</span></div>') +
        '<div class="draft-list">' +
          found.map(function (pair) {
            const level = (draft.confidence || {})[pair[0]] || "";
            return '<div class="draft-row"><span>' + escapeHtml(pair[1]) + '</span>' +
              '<strong>' + escapeHtml(String(draft[pair[0]])) + '</strong>' +
              (level === "low" ? '<em class="draft-uncertain">확인 필요</em>' : '') + '</div>';
          }).join("") +
        '</div>' +
        (missing.length
          ? '<p class="field-help" style="margin-top:12px">읽지 못한 항목: ' +
            escapeHtml(missing.map(function (pair) { return pair[1]; }).join(", ")) + '</p>'
          : '') +
        '<details class="draft-excerpt"><summary>공문에서 읽은 원문 보기</summary>' +
          '<pre>' + escapeHtml(excerpt) + '</pre></details>';
      showModal(
        "공문 판독 결과",
        "읽어낸 값은 초안입니다. 반드시 확인해주세요.",
        body,
        '<button type="button" class="btn btn-secondary" data-modal-close>취소</button>' +
        (found.length ? '<button type="button" class="btn btn-primary" data-action="apply-document-draft">이 내용으로 채우기</button>' : ''),
        false,
        "doc-draft-modal"
      );
    }

    /** 사용자가 확인한 판독 결과를 1단계 입력칸에 반영합니다. */
    function applyDocumentDraft() {
      const draft = STATE.pendingDocumentDraft || {};
      closeModal();
      const setFromDraft = function (id, value) {
        const el = document.getElementById(id);
        if (!el || !value) return false;
        el.value = value;
        return true;
      };

      let applied = 0;

      // 교육 종류
      if (draft.courseType) {
        const radio = document.querySelector('input[name="courseType"][value="' + draft.courseType + '"]');
        if (radio && !radio.checked) {
          radio.checked = true;
          toggleCreateCourseType(draft.courseType);
          applied++;
        }
      }
      if (setFromDraft("courseGeneration", draft.generation)) applied++;
      if (setFromDraft("courseTitle", draft.title)) applied++;
      if (setFromDraft("courseManagerName", draft.managerName)) applied++;
      if (setFromDraft("courseManagerPhone", draft.managerPhone)) applied++;
      if (setFromDraft("courseVenue", draft.venue)) applied++;
      if (setFromDraft("courseStartTime", draft.startTime)) applied++;
      if (setFromDraft("courseEndTime", draft.endTime)) applied++;
      if (setFromDraft("courseTargetAudience", draft.audience)) applied++;
      if (setFromDraft("courseCapacity", draft.capacity)) applied++;
      if (setFromDraft("courseFee", draft.fee)) applied++;
      if (setFromDraft("courseApplyDueDate", koreanDateTextToIso(draft.applyDueDate))) applied++;

      // 일수 · 날짜
      const draftDates = Array.isArray(draft.dates) && draft.dates.length
        ? draft.dates
        : [draft.day1Date, draft.day2Date].filter(Boolean);
      const days = clamp(Number(draft.courseDays || draftDates.length || 1), 1, 5);
      const daysSelect = document.getElementById("courseDays");
      if (daysSelect && Number(daysSelect.value || 0) !== days) {
        daysSelect.value = String(days);
        renderCreateDateAndMealFields();
        applied++;
      }
      draftDates.slice(0, days).forEach(function (dateText, index) {
        const day = index + 1;
        const iso = koreanDateTextToIso(dateText);
        if (!iso) return;
        const dateInput = document.getElementById("courseDate" + day);
        if (!dateInput) return;
        dateInput.value = iso;
        syncHiddenToKoreanDate(day);
        applied++;
      });

      showToast(
        applied ? applied + "개 항목에 공문 내용을 반영했습니다." : "반영할 수 있는 항목이 없었습니다.",
        applied ? "값이 맞는지 확인한 뒤 다음 단계로 넘어가주세요." : "필요한 항목은 직접 입력해주세요.",
        applied ? "success" : "info"
      );
      updateCreateSummary();
      // 비동기 화면 갱신이 뒤늦게 일어나도 방금 적용한 공문 값이 이전 초안에 덮이지 않게 즉시 저장합니다.
      saveCreateDraft();
      STATE.pendingDocumentDraft = null;
    }

    /** 'yyyy년 M월 d일' → 'yyyy-MM-dd' */
    function koreanDateTextToIso(value) {
      const match = String(value || "").match(/(\d{4})\s*년\s*(\d{1,2})\s*월\s*(\d{1,2})\s*일/);
      if (!match) return "";
      return match[1] + "-" + pad(Number(match[2])) + "-" + pad(Number(match[3]));
    }

    function renderCreateDateAndMealFields() {
      const dayInput = document.getElementById("courseDays");
      if (!dayInput) return;
      const days = clamp(Number(dayInput.value || 3), 1, 5);
      const datesRoot = document.getElementById("courseDateFields");
      const previousDates = {};
      const previousMeals = {};
      if (datesRoot) {
        datesRoot.querySelectorAll('input[type="date"]').forEach(function (input) {
          previousDates[input.name] = input.value;
        });
      }
      const mealRoot = document.getElementById("courseMealSlots");
      if (mealRoot) {
        mealRoot.querySelectorAll(".meal-checkbox").forEach(function (input) {
          previousMeals[input.value] = input.checked;
        });
      }
      if (datesRoot) {
        const firstDate = previousDates.courseDate1 || defaultCourseDate(0);
        datesRoot.innerHTML = Array.from({ length: days }, function (_, index) {
          const day = index + 1;
          const value = previousDates["courseDate" + day] || addDaysToDateInput(firstDate, index);
          return koreanDateField(day, value);
        }).join("");
        STATE.createDatesInitialized = true;
      }
      renderMealSlotTiles(days, previousMeals);
      syncCreatePeriodDefaults(false);
    }

    /** 식사 타일 — 실제 날짜를 함께 보여줍니다. */
    function renderMealSlotTiles(days, previousMeals) {
      const mealRoot = document.getElementById("courseMealSlots");
      if (!mealRoot) return;
      const saved = previousMeals || {};
      const hasSavedSelection = Object.keys(saved).length > 0;
      const lodgingType = (document.querySelector('input[name="lodgingType"]:checked') || {}).value || "stay";
      function isSelected(day, type) {
        const key = day + "|" + type;
        if (saved[key] != null) return Boolean(saved[key]);
        if (hasSavedSelection) return false;
        if (type === "lunch") return true;
        if (type === "dinner") return lodgingType === "stay" && day < days;
        return false;
      }
      function tile(day, type) {
        const key = day + "|" + type;
        return '<label class="check-tile meal-tile">' +
          '<input class="meal-checkbox" type="checkbox" value="' + key + '"' +
            (isSelected(day, type) ? " checked" : "") + '>' +
          '<span><b>' + escapeHtml(mealTypeLabel(type)) + '</b><small>' + escapeHtml(mealTypeTimeHint(type)) + '</small></span></label>';
      }
      mealRoot.style.setProperty("--meal-day-count", String(days));
      mealRoot.innerHTML = Array.from({ length: days }, function (_, index) {
        const day = index + 1;
        return '<article class="meal-day-card"><div class="meal-day-head"><span>' + day + '</span><div><strong>' +
          day + '일차</strong><small>' + escapeHtml(shortMealDateLabel(day)) + '</small></div></div><div class="meal-day-options">' +
          tile(day, "breakfast") + tile(day, "lunch") + tile(day, "dinner") + '</div></article>';
      }).join("");
      updateMealSelectionSummary();
    }

    function mealTypeLabel(type) {
      return type === "breakfast" ? "아침" : (type === "dinner" ? "저녁" : "점심");
    }

    function mealTypeTimeHint(type) {
      return type === "breakfast" ? "오전" : (type === "dinner" ? "오후·저녁" : "정오");
    }

    /** 해당 일차의 날짜를 'X월 X일'로. 아직 안 정했으면 'N일차'로 대체합니다. */
    function shortMealDateLabel(day) {
      const value = valueOf("courseDate" + day);
      const parts = String(value || "").match(/^(\d{4})-(\d{2})-(\d{2})$/);
      if (!parts) return day + "일차";
      return Number(parts[2]) + "월 " + Number(parts[3]) + "일";
    }

    function collectCourseContacts(includeEmpty) {
      const primary = document.querySelector('input[name="coursePrimaryContact"]:checked');
      const primaryIndex = primary ? Number(primary.value) : 0;
      return Array.from(document.querySelectorAll("#courseContactRows .contact-person-card[data-contact-index]")).map(function (row, order) {
        const nameInput = row.querySelector("[data-contact-name]");
        const phoneInput = row.querySelector("[data-contact-phone]");
        const index = Number(row.dataset.contactIndex || order);
        return {
          name: String(nameInput && nameInput.value || "").trim(),
          phone: String(phoneInput && phoneInput.value || "").trim(),
          primary: index === primaryIndex,
          index: index
        };
      }).filter(function (contact) {
        return includeEmpty || contact.name || contact.phone;
      });
    }

    function renderCourseContactRows(contacts) {
      const root = document.getElementById("courseContactRows");
      if (!root) return;
      const rows = Array.isArray(contacts) && contacts.length ? contacts.slice(0, 6) : [
        { name: "", phone: "", primary: true },
        { name: "", phone: "", primary: false }
      ];
      if (!rows.some(function (contact) { return contact.primary; })) rows[0].primary = true;
      root.innerHTML = rows.map(function (contact, index) {
        return renderCourseContactRow(contact, index, index > 0);
      }).join("");
      updateCreateSummary();
    }

    function addCourseContact() {
      const contacts = collectCourseContacts(true);
      if (contacts.length >= 6) {
        showToast("담당자는 최대 6명까지 추가할 수 있습니다.", "대표 문의 담당자는 한 명만 지정됩니다.", "error");
        return;
      }
      contacts.push({ name: "", phone: "", primary: false });
      renderCourseContactRows(contacts);
      const newInput = document.querySelector('#courseContactRows [data-contact-index="' + (contacts.length - 1) + '"] [data-contact-name]');
      if (newInput) newInput.focus();
      showToast("담당자 입력란을 추가했습니다.", "이름과 연락처를 함께 입력해주세요.");
    }

    function removeCourseContact(index) {
      const contacts = collectCourseContacts(true);
      const removed = contacts[Number(index)];
      contacts.splice(Number(index), 1);
      if (!contacts.length) contacts.push({ name: "", phone: "", primary: true });
      if (removed && removed.primary) {
        contacts.forEach(function (contact, order) { contact.primary = order === 0; });
        showToast("대표 문의 담당자를 변경했습니다.", "삭제 후 첫 번째 담당자를 대표 문의 담당자로 지정했습니다.");
      }
      renderCourseContactRows(contacts);
      scheduleCreateDraftSave();
    }

    function updateMealSelectionSummary() {
      const summary = document.getElementById("courseMealSelectionSummary");
      if (!summary) return;
      const skipped = checkedOf("courseMealSkip");
      const count = collectMealSlots(document).length;
      summary.classList.toggle("is-skipped", skipped);
      summary.classList.toggle("has-selection", !skipped && count > 0);
      summary.textContent = skipped
        ? "식사 자리 배치 없이 개설합니다. 나중에 운영 설정에서 다시 켤 수 있어요."
        : count
          ? "총 " + count + "끼 식사를 정해진 테이블에서 진행해요."
          : "아직 선택한 식사가 없습니다.";
    }

    function formatCreateDateSummary(value) {
      const match = String(value || "").match(/^(\d{4})-(\d{2})-(\d{2})$/);
      return match ? Number(match[2]) + "월 " + Number(match[3]) + "일" : "";
    }

    function updateCreateSummary() {
      const form = document.getElementById("courseCreateForm");
      if (!form) return;
      const courseType = (form.querySelector('[name="courseType"]:checked') || {}).value || "vup";
      const title = courseType === "general"
        ? valueOf("courseTitle")
        : valueOf("courseGeneration")
          ? "제" + valueOf("courseGeneration") + " 벤처투자 전문인력 양성과정(V-Up)"
          : "";
      const contacts = collectCourseContacts(false);
      const primary = contacts.find(function (contact) { return contact.primary; }) || contacts[0];
      const days = clamp(Number(valueOf("courseDays") || 3), 1, 5);
      const first = formatCreateDateSummary(valueOf("courseDate1"));
      const last = formatCreateDateSummary(valueOf("courseDate" + days));
      const meals = checkedOf("courseMealSkip") ? 0 : collectMealSlots(form).length;
      const set = function (id, value) { const el = document.getElementById(id); if (el) el.textContent = value; };
      set("createSummaryTitle", title || (courseType === "general" ? "교육명을 입력해주세요." : "기수를 입력해주세요."));
      set("createSummaryContact", primary ? primary.name + (primary.phone ? " · " + primary.phone : "") : "담당자를 확인해주세요.");
      set("createSummarySchedule", first ? first + (last && last !== first ? " ~ " + last : "") + " · " + days + "일" : days + "일 일정");
      set("createSummaryMeals", checkedOf("courseMealSkip") ? "배치하지 않음" : meals ? "총 " + meals + "끼" : "선택 전");
      set("createSummaryCode", valueOf("courseAccessCode") || "-");
      const step = clamp(Number(STATE.createCourseStep || 1), 1, 4);
      set("createSummaryProgressText", step + " / 4 단계 작성 중");
      const bar = document.getElementById("createSummaryProgressBar");
      if (bar) bar.style.width = (step * 25) + "%";
    }

    function regenerateCreateAccessCode() {

    /* 개설 위저드는 '일수 + 형태' 입력으로 바뀌었지만,
       강의 기본 정보 수정 모달은 아직 이 선택 목록을 씁니다. */
    function lodgingOptionMarkup(selectedDays) {
      const labels = {
        1: "당일치기 (1일)",
        2: "1박 2일",
        3: "2박 3일",
        4: "3박 4일",
        5: "4박 5일"
      };
      return [1, 2, 3, 4, 5].map(function (days) {
        return '<option value="' + days + '"' +
          (Number(selectedDays) === days ? " selected" : "") +
          '>' + labels[days] + '</option>';
      }).join("");
    }

    function addDaysToDateInput(value, dayOffset) {
      const parts = String(value || "").match(/^(\d{4})-(\d{2})-(\d{2})$/);
      if (!parts) return defaultCourseDate(dayOffset);
      const date = new Date(
        Number(parts[1]),
        Number(parts[2]) - 1,
        Number(parts[3])
      );
      date.setDate(date.getDate() + Number(dayOffset || 0));
      return date.getFullYear() + "-" +
        pad(date.getMonth() + 1) + "-" +
        pad(date.getDate());
    }

    function autofillCourseDatesFromFirst() {
      const first = document.getElementById("courseDate1");
      const dayInput = document.getElementById("courseDays");
      if (!first || !first.value || !dayInput) return;
      const days = clamp(Number(dayInput.value || 1), 1, 5);
      for (let day = 2; day <= days; day++) {
        const hidden = document.getElementById("courseDate" + day);
        if (!hidden) continue;
        hidden.value = addDaysToDateInput(first.value, day - 1);
        syncHiddenToKoreanDate(day);
      }
      syncCreatePeriodDefaults(true);
      refreshMealTileLabels();
    }

    /** 날짜가 바뀌면 식사 타일의 'X월 X일' 표기도 따라갑니다. */
    function refreshMealTileLabels() {
      const mealRoot = document.getElementById("courseMealSlots");
      if (!mealRoot) return;
      const saved = {};
      mealRoot.querySelectorAll(".meal-checkbox").forEach(function (input) {
        saved[input.value] = input.checked;
      });
      const dayInput = document.getElementById("courseDays");
      renderMealSlotTiles(clamp(Number(dayInput && dayInput.value || 3), 1, 5), saved);
    }

    function setDateTimeDefault(id, value, force) {
      const input = document.getElementById(id);
      if (!input) return;
      if (force || !input.value) input.value = value;
    }

    function dateInputAt(value, hour, minute) {
      const parts = String(value || "").match(/^(\d{4})-(\d{2})-(\d{2})$/);
      if (!parts) return "";
      const date = new Date(
        Number(parts[1]),
        Number(parts[2]) - 1,
        Number(parts[3]),
        Number(hour || 0),
        Number(minute || 0),
        0,
        0
      );
      return toLocalDateTimeString(date);
    }

    function shiftDateTimeLocal(value, dayOffset, hour, minute) {
      const date = new Date(value);
      if (isNaN(date.getTime())) return "";
      date.setDate(date.getDate() + Number(dayOffset || 0));
      date.setHours(Number(hour || 0), Number(minute || 0), 0, 0);
      return toLocalDateTimeString(date);
    }

    function syncCreatePeriodDefaults(force) {
      const dayInput = document.getElementById("courseDays");
      const firstDate = valueOf("courseDate1");
      if (!dayInput || !firstDate) return;
      const days = clamp(Number(dayInput.value || 1), 1, 5);
      const lastDate = valueOf("courseDate" + days) ||
        addDaysToDateInput(firstDate, days - 1);
      const courseStart = dateInputAt(firstDate, 8, 0);
      const courseEnd = dateInputAt(lastDate, 23, 59);
      setDateTimeDefault(
        "coursePreOpenAt",
        shiftDateTimeLocal(courseStart, -7, 9, 0),
        force
      );
      setDateTimeDefault(
        "coursePreCloseAt",
        shiftDateTimeLocal(courseStart, -1, 18, 0),
        force
      );
      setDateTimeDefault("courseOpenAt", courseStart, force);
      setDateTimeDefault("courseCloseAt", courseEnd, force);
    }

       라벨 옆에 짧은 경고를 답니다. 각 단계에서 다루는 칸만 먼저 비웁니다. */
    const CREATE_STEP_FIELDS = {
      1: ["courseGeneration", "courseTitle", "courseManagerName", "courseManagerPhone", "courseSubManagerName"],
      2: [],
      3: [],
      4: ["courseAccessCode", "coursePreOpenAt", "coursePreCloseAt", "courseOpenAt", "courseCloseAt"]

    function validateCreateStep(step) {
      const fields = CREATE_STEP_FIELDS[step] || [];
      clearFieldErrors(fields);
      let ok = true;

      if (step === 1) {
        const selectedType = document.querySelector('input[name="courseType"]:checked');
        const isGeneral = selectedType && selectedType.value === "general";

        if (isGeneral) {
          if (!valueOf("courseTitle")) { setFieldError("courseTitle", "교육명을 입력해주세요"); ok = false; }
        } else if (!validateGenerationInput()) {
          ok = false;
        }

        const contactRows = Array.from(document.querySelectorAll("#courseContactRows .contact-person-card[data-contact-index]"));
        const contacts = collectCourseContacts(true);
        contactRows.forEach(function (row) {
          row.classList.remove("is-invalid");
          const oldError = row.querySelector(".contact-row-error");
          if (oldError) oldError.remove();
        });
        contacts.forEach(function (contact, index) {
          let message = "";
          if (index === 0 && !contact.name && !contact.phone) message = "기본 담당자 이름과 연락처를 입력해주세요.";
          else if (Boolean(contact.name) !== Boolean(contact.phone)) message = "이름과 연락처를 함께 입력해주세요.";
          else if (contact.phone && !/^\d{2,4}-\d{3,4}-\d{4}$/.test(contact.phone)) message = "하이픈을 포함한 연락처 형식을 확인해주세요.";
          if (message) {
            const row = contactRows[index];
            if (row) {
              row.classList.add("is-invalid");
              row.insertAdjacentHTML("beforeend", '<span class="contact-row-error" role="alert">' + escapeHtml(message) + '</span>');
            }
            ok = false;
          }
        });
        const filledContacts = contacts.filter(function (contact) { return contact.name && contact.phone; });
        const primaryContacts = filledContacts.filter(function (contact) { return contact.primary; });
        if (!filledContacts.length) {
          showToast("문의 담당자를 입력해주세요.", "이름과 연락처가 모두 필요합니다.", "error");
          ok = false;
        } else if (primaryContacts.length !== 1) {
          showToast("대표 문의 담당자를 한 명 지정해주세요.", "사용자 페이지 최상단에 노출할 담당자는 한 명만 선택할 수 있습니다.", "error");
          ok = false;
        }

        updateManagerProfileMismatchNotice();
        if (!ok) {
          const firstContactError = document.querySelector("#courseContactRows .is-invalid input");
          if (firstContactError) firstContactError.focus();
          else focusFirstInvalid(fields);
        }
        return ok;
      }

      if (step === 2) {
        const days = Number(valueOf("courseDays") || 0);
        if (!days) {
          showToast("교육 일수를 선택해주세요.", "오프라인 교육 일수는 1~5일 사이로 설정할 수 있습니다.", "error");
          return false;
        }
        const dateIds = [];
        for (let day = 1; day <= days; day++) dateIds.push("courseDate" + day);
        clearFieldErrors(dateIds);
        for (let day = 1; day <= days; day++) {
          if (!valueOf("courseDate" + day)) {
            setFieldError("courseDate" + day, "날짜를 입력해주세요");
            ok = false;
          }
        }
        if (!ok) focusFirstInvalid(dateIds);
        return ok;
      }

      if (step === 3) {
        // '배치가 필요하지 않은 교육'을 고르면 식사 선택 없이 넘어갑니다.
        if (checkedOf("courseMealSkip")) {
          const skipRoot = document.getElementById("courseMealSlots");
          if (skipRoot) skipRoot.classList.remove("meal-slot-invalid");
          return true;
        }
        if (!collectMealSlots(document).length) {
          const root = document.getElementById("courseMealSlots");
          if (root) root.classList.add("meal-slot-invalid");
          showToast("식사 일정을 선택해주세요.", "사용자에게 안내할 식사를 하나 이상 선택해야 합니다.", "error");
          return false;
        }
        const root = document.getElementById("courseMealSlots");
        if (root) root.classList.remove("meal-slot-invalid");
        return true;
      }

      if (step === 4) {
        if (!/^\d{4}$/.test(valueOf("courseAccessCode"))) {
          setFieldError("courseAccessCode", "숫자 4자리여야 합니다");
          ok = false;
        }
        const periods = [
          ["coursePreOpenAt", "사전 안내 시작"],
          ["coursePreCloseAt", "사전 안내 종료"],
          ["courseOpenAt", "현장 안내 시작"],
          ["courseCloseAt", "현장 안내 종료"]
        ];
        periods.forEach(function (item) {
          if (!valueOf(item[0])) { setFieldError(item[0], "입력해주세요"); ok = false; }
        });
        if (ok) {
          const preStart = new Date(valueOf("coursePreOpenAt")).getTime();
          const preEnd = new Date(valueOf("coursePreCloseAt")).getTime();
          const courseStart = new Date(valueOf("courseOpenAt")).getTime();
          const courseEnd = new Date(valueOf("courseCloseAt")).getTime();
          if (!(preStart < preEnd)) { setFieldError("coursePreCloseAt", "시작보다 뒤여야 합니다"); ok = false; }
          if (!(preEnd <= courseStart)) { setFieldError("courseOpenAt", "사전 안내 종료 이후여야 합니다"); ok = false; }
          if (!(courseStart < courseEnd)) { setFieldError("courseCloseAt", "시작보다 뒤여야 합니다"); ok = false; }
        }
        if (!ok) focusFirstInvalid(fields);
        return ok;
      }

      return true;
    }

    function openCourse(siteId, tab, navigationOptions) {
      const navigation = navigationOptions || {};
      const requestVersion = ++STATE.navigationVersion;
      STATE.view = "course";
      updateNavigationChrome("course");
      // 다른 목록의 아래쪽에서 강의를 열었더라도 강의 화면은 제목부터 보입니다.

        .then(function (result) {
          if (
            requestVersion !== STATE.navigationVersion ||
            STATE.view !== "course"
          ) return result;
          STATE.course = result.course;
          STATE.pageBlocks = JSON.parse(JSON.stringify(
            (result.course.settings && result.course.settings.pageBlocks) || []
          ));
          STATE.pageBuilderDirty = false;
          STATE.editingPageBlockIndex = -1;
          STATE.pendingPageBlockFile = null;
          STATE.operationsPlan = JSON.parse(JSON.stringify(
            (result.course.settings && result.course.settings.operationsPlan) || getEmptyOperationsPlan()
          ));
          STATE.operationStageFilter = "all";
          STATE.segmentsLoaded = false;
          STATE.seatMapLoaded = false;
          STATE.segments = [];
          STATE.segmentSummary = "";
          STATE.seatMap = null;
          STATE.roomPlan = null;
          STATE.materials = result.materials || [];
          STATE.photos = result.photos || [];
          STATE.surveys = result.surveys || [];
          STATE.resourceLoaded = { materials: false, photos: false, surveys: false };
          STATE.roster = null;
          STATE.rosterRoomDraft = null;
          STATE.photoNextPageToken = "";
          STATE.photoHasMore = false;
          STATE.health = result.health || {};
          STATE.operations = result.operations || { logs: [] };
          STATE.currentCourseTab = tab || STATE.currentCourseTab || "overview";
          if (!navigation.fromHistory) {
            writePortalHistory("course", {
              courseId: STATE.course.siteId,
              tab: STATE.currentCourseTab
            });
          }
          STATE.selectedFiles = { material: [], photo: [] };
          STATE.mealInput = {
            participants: [],
            seatGrid: [],
            roommateRows: [],
            rosterName: "",
            seatName: "",
            roommateName: ""
          };
          updateNavigationChrome("course");
          renderCourseDetail();
          return loadCourseTabResources(STATE.currentCourseTab).then(function () { return result; });
        })
        .catch(function (error) {
          if (
            requestVersion !== STATE.navigationVersion ||
            STATE.view !== "course"
          ) return;
          showToast("강의를 열지 못했습니다.", error.message, "error");
          if (isContentEmpty()) renderFatalError("강의 화면을 열지 못했습니다.", error.message);
        })
        .finally(hideLoader);
    }

    function ensureOperationsPlan() {
      const source = STATE.operationsPlan && typeof STATE.operationsPlan === "object"
        ? STATE.operationsPlan : {};
      const defaults = getEmptyOperationsPlan();
      const saved = Array.isArray(source.tasks) ? source.tasks : [];
      const savedMap = {};
      saved.forEach(function (task) { if (task && task.id) savedMap[task.id] = task; });
      const tasks = defaults.tasks.map(function (task) {
        return Object.assign({}, task, savedMap[task.id] || {}, { system: true });
      });
      saved.forEach(function (task) {
        if (task && task.id && !tasks.some(function (item) { return item.id === task.id; })) {
          tasks.push(Object.assign({}, task, { system: false }));
        }
      });
      STATE.operationsPlan = Object.assign({}, defaults, source, {
        funnel: Object.assign({}, defaults.funnel, source.funnel || {}),
        tasks: tasks
      });
      return STATE.operationsPlan;
    }

    function getOperationStageMeta(key) {
      return OPERATION_STAGE_META.find(function (item) { return item.key === key; }) || OPERATION_STAGE_META[0];
    }

    function todayDateKey() {
      const date = new Date();
      return date.getFullYear() + "-" + String(date.getMonth() + 1).padStart(2, "0") + "-" + String(date.getDate()).padStart(2, "0");
    }

    function formatShortDate(value) {
      const text = String(value || "");
      return /^\d{4}-\d{2}-\d{2}$/.test(text) ? text.slice(5).replace("-", ".") : text;
    }

    function getOperationsSummaryUi(plan) {
      const tasks = plan.tasks || [];
      const today = todayDateKey();
      const soon = shiftIsoDate(today, 7);
      const pending = tasks.filter(function (task) { return !task.done; });
      return {
        total: tasks.length,
        done: tasks.filter(function (task) { return task.done; }).length,
        pending: pending.length,
        overdue: pending.filter(function (task) { return task.dueDate && task.dueDate < today; }).length,
        dueSoon: pending.filter(function (task) { return task.dueDate && task.dueDate >= today && task.dueDate <= soon; }).length,
        percent: tasks.length ? Math.round(tasks.filter(function (task) { return task.done; }).length / tasks.length * 100) : 0
      };
    }

    function submitOperationsFunnelForm() {
      const plan = ensureOperationsPlan();
      const numbers = {
        applied: Math.max(0, Number(valueOf("operationApplied") || 0)),
        eligible: Math.max(0, Number(valueOf("operationEligible") || 0)),
        approved: Math.max(0, Number(valueOf("operationApproved") || 0)),
        paymentRequested: Math.max(0, Number(valueOf("operationPaymentRequested") || 0)),
        paid: Math.max(0, Number(valueOf("operationPaid") || 0))
      };
      if (numbers.eligible > numbers.applied || numbers.approved > numbers.eligible ||
          numbers.paymentRequested > numbers.approved || numbers.paid > numbers.paymentRequested) {
        return showToast("인원 현황의 순서를 확인해주세요.", "신청 ≥ 자격 확인 ≥ 승인 ≥ 결제 안내 ≥ 결제 완료여야 합니다.", "error");
      }
      const registrationUrl = valueOf("operationRegistrationUrl").trim();
      if (registrationUrl && !isSafeUserPageHttpsUrl(registrationUrl)) {
        return showToast("안전한 수강신청 주소를 입력해주세요.", "https:// 주소만 저장할 수 있습니다.", "error");
      }
      const openAt = valueOf("operationRegistrationOpenAt");
      const closeAt = valueOf("operationRegistrationCloseAt");
      if (openAt && closeAt && new Date(openAt).getTime() >= new Date(closeAt).getTime()) {
        return showToast("접수 기간을 확인해주세요.", "접수 시작은 마감보다 앞서야 합니다.", "error");
      }
      plan.registrationUrl = registrationUrl;
      plan.registrationOpenAt = openAt;
      plan.registrationCloseAt = closeAt;
      plan.funnel = numbers;
      closeModal(true);
      persistOperationsPlan("수강신청·결제 현황을 저장했습니다.");
    }

    function applyRecommendedOperationDeadlines() {
      const course = STATE.course || {};
      const start = toDateInput(course.day1Date || "");
      if (!start) return showToast("교육 일정을 먼저 입력해주세요.", "추천 마감일은 교육 시작일을 기준으로 계산합니다.", "error");
      const offsets = {
        "recruitment-details": -35, "recruitment-page": -31, "recruitment-open": -30,
        "selection-eligibility": -21, "selection-approval": -18, "selection-payment-guide": -17, "selection-payment-confirm": -12,
        "preparation-plan": -14, "preparation-instructor": -14, "preparation-supplies": -7, "preparation-textbook": -7,

        "followup-expense": 5, "followup-report": 7, "followup-insight": 10
      };
      let changed = 0;
      (ensureOperationsPlan().tasks || []).forEach(function (task) {
        if (!task.dueDate && Object.prototype.hasOwnProperty.call(offsets, task.id)) {
          task.dueDate = shiftIsoDate(start, offsets[task.id]);
          changed++;
        }
      });
      if (!changed) return showToast("이미 모든 표준 업무에 마감일이 있습니다.", "기존 날짜는 덮어쓰지 않았습니다.");
      persistOperationsPlan("교육 시작일 기준 추천 마감일 " + changed + "건을 채웠습니다.");
    }

    function persistOperationsPlan(successMessage) {
      if (!STATE.course || STATE.operationSaving) return Promise.resolve();
      STATE.operationSaving = true;
      updateCourseTabContent();
      return api("portalSaveCourseOperationsPlan", STATE.course.siteId, ensureOperationsPlan(), STATE.course.revision)
        .then(function (result) {
          STATE.operationsPlan = JSON.parse(JSON.stringify(result.operationsPlan || getEmptyOperationsPlan()));
          STATE.course.settings.operationsPlan = STATE.operationsPlan;
          STATE.course.operationsSummary = result.operationsSummary || getOperationsSummaryUi(STATE.operationsPlan);
          STATE.course.revision = Number(result.revision || STATE.course.revision || 0);
          STATE.dashboardFetchedAt = 0;
          STATE.courseListCache = {};
          showToast(successMessage || "운영 업무를 저장했습니다.", "운영 홈과 강의 준비도에 반영되었습니다.");
          return result;
        })
        .catch(function (error) {
          showToast("운영 업무를 저장하지 못했습니다.", error.message, "error");
          return openCourse(STATE.course.siteId, "operations");
        })
        .finally(function () {
          STATE.operationSaving = false;
          if (STATE.currentCourseTab === "operations") updateCourseTabContent();
        });
    }

    function loadCourseRoster(forceRefresh) {
      if (!STATE.course) return Promise.resolve();
      STATE.rosterLoading = true;
      updateCourseTabContent();
      return api("portalGetCourseRoster", STATE.course.siteId, Boolean(forceRefresh))
        .then(function (result) {
          STATE.rosterLoading = false;
          // 명부는 '교육생 정보'뿐 아니라 '교육생·식사' 탭에서도 씁니다.
          if (!STATE.course ||
              ["roster", "meal"].indexOf(STATE.currentCourseTab) === -1) return result;
          STATE.roster = result;
          STATE.rosterRoomDraft = null;
          updateCourseTabContent();
          return result;
        })
        .catch(function (error) {
          STATE.rosterLoading = false;
          STATE.roster = { error: error.message };
          updateCourseTabContent();
        });
    }

    function filterRoster(roster) {
      const filter = STATE.rosterFilter || "all";
      const query = String(STATE.rosterSearch || "").replace(/\s/g, "").toLowerCase();
      return roster.filter(function (item) {
        if (filter === "pending" && item.status === "확인완료") return false;
        if (filter === "reviewPending" && item.reviewStatus && item.reviewStatus !== "pending") return false;
        if (filter === "paymentPending" && ["requested", "transfer-pending"].indexOf(item.paymentStatus) === -1) return false;
        if (filter === "insurancePending" && item.insuranceStatus !== "pending") return false;
        if (filter === "noGender" && item.gender) return false;
        if (filter === "noRoom" && String(item.room || "").trim()) return false;
        if (filter === "excluded" && !item.excluded) return false;
        if (filter === "single" && item.roomType !== "single") return false;
        if (!query) return true;
        const haystack = [item.name, item.organization, item.appliedOrganization,
          item.position, item.appliedPosition, item.room, item.reviewStatus, item.paymentStatus]
          .join("").replace(/\s/g, "").toLowerCase();
        return haystack.indexOf(query) !== -1;
      });
    }

    function saveSeatingFlag(key, changes, quiet) {
      if (!STATE.course) return Promise.resolve();
      return api("portalSetRosterSeatingFlags", STATE.course.siteId,
        [Object.assign({ key: key }, changes)])
        .then(function (result) {
          if (!quiet && result.changed) showToast("저장했습니다.", result.message, "info");
          const item = ((STATE.roster && STATE.roster.roster) || [])
            .filter(function (r) { return r.key === key; })[0];
          if (item) {
            const wasExcluded = Boolean(item.excluded);
            Object.assign(item, changes);
            const summary = STATE.roster && STATE.roster.summary;
            if (summary && changes.excluded !== undefined && wasExcluded !== Boolean(changes.excluded)) {
              summary.excluded = Math.max(0, Number(summary.excluded || 0) + (changes.excluded ? 1 : -1));
              summary.seatingTarget = Math.max(0, Number(summary.total || 0) - summary.excluded);
            }
            if (summary && result.mealAssignment) summary.mealAssignment = result.mealAssignment;
          }
          return result;
        })
        .catch(showErrorToast);
    }

    const ROSTER_COLUMN_RULES = [
      ["name", /^(이름|성명|성함|참가자|교육생|name)$/i],
      ["phone", /(휴대|연락처|전화|핸드폰|hp|phone|mobile)/i],
      ["organization", /(소속|기관|회사|업체|조합|company|org)/i],
      ["position", /(직책|직급|직위|position|title)/i],
      ["gender", /^(성별|gender|sex)$/i],
      ["birthYear", /(생년|출생|나이|birth|year)/i],
      ["roomType", /(객실|숙박|룸타입|1인실|room)/i],
      ["room", /^(방|방번호|호수|객실번호)$/i]
    ];

    function handleRosterFile(file) {
      if (!STATE.course) return;
      if (file.size > 8 * 1024 * 1024) {
        showToast("파일이 너무 큽니다.", "8MB 이하 엑셀·CSV로 올려주세요.", "error");
        return;
      }
      showLoader("명단 읽는 중", file.name + "의 열 이름을 확인하고 있습니다.");
      ensureXlsxLoaded()
        .then(function (XLSX) { return readWorkbookFile(XLSX, file, "roster"); })
        .then(function (workbookData) {
          const parsed = parseRosterWorkbook(trimEmptyGrid(workbookData.rows));
          if (!parsed.rows.length) {
            throw new Error("이름과 연락처 열을 찾지 못했습니다. 첫 행에 열 이름이 있는지 확인해주세요.");
          }
          showRosterImportModal(parsed);
        })
        .catch(showErrorToast)
        .finally(hideLoader);
    }

    /** 머리글 행을 찾아 열을 짝지어 줍니다. */
    function parseRosterWorkbook(grid) {
      let headerIndex = -1;
      let mapping = null;
      for (let i = 0; i < Math.min(grid.length, 12); i++) {
        const candidate = mapRosterColumns(grid[i]);
        if (candidate && candidate.name !== undefined && candidate.phone !== undefined) {
          headerIndex = i;
          mapping = candidate;
          break;
        }
      }
      if (!mapping) return { rows: [], mapping: null, headerIndex: -1 };

      const rows = [];
      for (let i = headerIndex + 1; i < grid.length; i++) {
        const cells = grid[i];
        const pick = function (field) {
          const col = mapping[field];
          return col === undefined ? "" : String(cells[col] || "").trim();
        };
        const name = pick("name");
        if (!name) continue;
        rows.push({
          name: name,
          phone: pick("phone"),
          organization: pick("organization"),
          position: pick("position"),
          gender: pick("gender"),
          birthYear: pick("birthYear"),
          roomType: pick("roomType"),
          room: pick("room")
        });
      }
      return { rows: rows, mapping: mapping, headerIndex: headerIndex };
    }

    function mapRosterColumns(headerRow) {
      const found = {};
      (headerRow || []).forEach(function (cell, index) {
        const text = String(cell || "").replace(/\s/g, "");
        if (!text) return;
        ROSTER_COLUMN_RULES.forEach(function (rule) {
          if (found[rule[0]] !== undefined) return;
          if (rule[1].test(text)) found[rule[0]] = index;
        });
      });
      return Object.keys(found).length ? found : null;
    }

          ? '<div class="room-issue"><h4>한 방에 3명 이상</h4><ul>' +
            v.overfilled.map(function (item) {
              return '<li><b>' + escapeHtml(item.room) + '호</b> · ' +
                escapeHtml(item.names.join(", ")) + '</li>';
            }).join("") + '</ul></div>'
          : "") +
        (v.lost && v.lost.length
          ? '<div class="room-issue"><h4>방 배정이 없는 교육생 ' + v.lost.length + '명</h4>' +
            '<p>' + escapeHtml(v.lost.join(", ")) + '</p></div>'
          : "") +
        (v.unknown && v.unknown.length
          ? '<div class="room-issue"><h4>명부에 없는 항목</h4><p>' +
            escapeHtml(v.unknown.join(", ")) + '</p></div>'
          : "");
      showModal(
        result.blocked ? "저장할 수 없습니다" : "이대로 저장할까요?",
        result.blocked ? "아래 문제를 먼저 정리해주세요." : "배정이 비는 교육생이 있습니다.",
        body,
        '<button type="button" class="btn btn-secondary" data-modal-close>닫기</button>' +
          (result.blocked ? "" :
            '<button type="button" class="btn btn-primary" data-confirm-room-save>그대로 저장</button>'),
        true
      );
    }

    /* ==========================================================
     * 자리·방 배정 — 한 번에 한 과업만
     * ----------------------------------------------------------
     * 세 카드를 한꺼번에 펼쳐 두면 "뭐부터 하라는 거지?"가 됩니다.
     * 지금 할 단계만 펼치고, 끝난 단계는 한 줄로 접습니다.
     * 각 단계는 [자동] / [내가 만든 표] 두 갈래를 나란히 둡니다.
     * ========================================================== */
    const SEATING_STEPS = [
      { key: "seat", no: "1", title: "강의장 좌석", hint: "가나다순으로 자리 정하기" },
      { key: "room", no: "2", title: "숙박 방 배정", hint: "성별로 층 나누고 짝 맞추기" },
      { key: "meal", no: "3", title: "식사 자리", hint: "끼니마다 테이블 나누기" }
    ];

    function seatingStepState(key) {
      const summary = (STATE.roster && STATE.roster.summary) || null;
      const roster = (STATE.roster && STATE.roster.roster) || [];
      if (key === "seat") {
        const seated = roster.filter(function (r) { return r.classSeat; }).length;
        if (STATE.seatMap) {
          return { done: !STATE.seatMap.dirty, label: STATE.seatMap.cols + "열 × " +
            STATE.seatMap.rows + "행 · " + (STATE.seatMap.report && STATE.seatMap.report.seated || 0) + "명" };
        }
        return { done: seated > 0, label: seated ? seated + "명 배치됨" : "" };
      }
      if (key === "room") {
        const roomed = roster.filter(function (r) { return String(r.room || "").trim(); }).length;
        if (STATE.roomPlan) {
          return { done: !STATE.roomPlan.dirty,
            label: (STATE.roomPlan.report && STATE.roomPlan.report.total || 0) + "명 · " +
              ((STATE.roomPlan.report && STATE.roomPlan.report.maleRooms || 0) +
               (STATE.roomPlan.report && STATE.roomPlan.report.femaleRooms || 0) +
               (STATE.roomPlan.report && STATE.roomPlan.report.mixedGenderRooms || 0)) + "실" };
        }
        return { done: roomed > 0, label: roomed ? roomed + "명 배정됨" : "" };
      }
      const mealAssignment = summary && summary.mealAssignment;
      return {
        done: Boolean(mealAssignment && mealAssignment.completed && !mealAssignment.stale),
        label: mealAssignment && mealAssignment.stale
          ? "입력 정보 변경됨 · 재배치 필요"
          : mealAssignment && mealAssignment.completed
            ? Number(mealAssignment.participantCount || 0) + "명 · " +
              Number(mealAssignment.mealCount || 0) + "끼니 저장됨"
          : ""
      };
    }

    function currentSeatingStep() {
      if (STATE.seatingStep) return STATE.seatingStep;
      for (let i = 0; i < SEATING_STEPS.length; i++) {
        if (!seatingStepState(SEATING_STEPS[i].key).done) return SEATING_STEPS[i].key;
      }
      return SEATING_STEPS[SEATING_STEPS.length - 1].key;
    }

    function renderMealTab() {
      // 명부를 아직 안 읽었다면 먼저 읽습니다(다른 탭을 거치지 않아도 되게).
      if (!STATE.roster && !STATE.rosterLoading) {
        loadCourseRoster(false);
        return renderTabLoading("교육생 명부를 불러오는 중입니다.");
      }
      if (STATE.rosterLoading) return renderTabLoading("교육생 명부를 불러오는 중입니다.");

      const summary = (STATE.roster && STATE.roster.summary) || null;
      loadSavedSeatMap();

      if (!summary || Number(summary.total || 0) === 0) {
        return '<div class="panel panel-pad">' +
          renderAsk(
            "아직 교육생 명단이 없어요.",
            "명단이 있어야 좌석·방·식사를 배정할 수 있습니다. 어떻게 할까요?",
            [
              { label: "엑셀로 명단 올리기", hint: "신청 파일 그대로", primary: true, tab: "roster" },
              { label: "연결 시트에서 가져오기", hint: "구글 시트 연동", tab: "roster" }
            ]
          ) + '</div>';
      }

      const active = currentSeatingStep();
      return '<div class="task-flow">' +
        SEATING_STEPS.map(function (step) {
          return renderSeatingStep(step, active, summary);
        }).join("") +
      '</div>';
    }

    function renderSeatingStep(step, active, summary) {
      const state = seatingStepState(step.key);
      const isActive = step.key === active;
      const cls = "task-step" + (isActive ? " is-active" : "") + (state.done ? " is-done" : "");

      if (!isActive) {
        return '<button type="button" class="' + cls + ' task-step-collapsed"' +
          ' data-seating-step="' + step.key + '" id="step-' + step.key + '">' +
          '<span class="task-step-mark">' + (state.done ? "✓" : step.no) + '</span>' +
          '<span class="task-step-copy"><strong>' + escapeHtml(step.title) + '</strong>' +
            '<span>' + escapeHtml(state.label || step.hint) + '</span></span>' +
          '<span class="task-step-go">' + (state.done ? "수정" : "시작") + ' →</span>' +
        '</button>';
      }

      return '<section class="' + cls + '" id="step-' + step.key + '">' +
        '<header class="task-step-head">' +
          '<span class="task-step-mark">' + (state.done ? "✓" : step.no) + '</span>' +
          '<span class="task-step-copy"><strong>' + escapeHtml(step.title) + '</strong>' +
            '<span>' + escapeHtml(step.hint) + '</span></span>' +
          (state.done ? '<span class="task-step-badge">완료</span>' : '') +
        '</header>' +
        '<div class="task-step-body">' +
          (step.key === "seat" ? renderSeatStepBody(summary)
            : step.key === "room" ? renderRoomStepBody(summary)
            : renderMealStepBody(summary)) +
        '</div>' +
        renderStepFooter(step, state) +
      '</section>';
    }

    /** 단계 하단: 다음으로 넘어가는 길을 항상 보여줍니다. */
    function renderStepFooter(step, state) {
      const index = SEATING_STEPS.findIndex(function (x) { return x.key === step.key; });
      const next = SEATING_STEPS[index + 1];
      return '<footer class="task-step-foot">' +
        (index > 0
          ? '<button type="button" class="btn btn-text" data-seating-step="' +
            SEATING_STEPS[index - 1].key + '">← ' + SEATING_STEPS[index - 1].title + '</button>'
          : '<span></span>') +
        (next
          ? '<button type="button" class="btn ' + (state.done ? "btn-primary" : "btn-secondary") +
            '" data-seating-step="' + next.key + '">' +
            (state.done ? "다음: " + next.title : next.title + " 먼저 하기") + ' →</button>'
          : (state.done
              ? '<span class="task-step-done-note">세 단계를 모두 마쳤습니다.</span>'
              : '<span></span>')) +
      '</footer>';
    }

    /* ---------- 공통: 대화형 질문 ---------- */
    function renderAsk(title, question, options) {
      return '<div class="ask">' +
        '<div class="ask-bubble">' +
          '<strong>' + escapeHtml(title) + '</strong>' +
          (question ? '<span>' + escapeHtml(question) + '</span>' : '') +
        '</div>' +
        '<div class="ask-options">' +
          options.map(function (opt, index) {
            const attrs = [];
            if (opt.action) attrs.push('data-action="' + opt.action + '"');
            if (opt.tab) attrs.push('data-course-tab="' + opt.tab + '"');
            if (opt.mode) attrs.push('data-seating-mode="' + opt.mode + '"');
            if (opt.disabled) attrs.push('disabled');
            return '<button type="button" class="ask-option' +
              (opt.primary ? " is-primary" : "") + '" ' + attrs.join(" ") + '>' +
              '<span class="ask-option-no">' + (index + 1) + '</span>' +
              '<span class="ask-option-copy"><strong>' + escapeHtml(opt.label) + '</strong>' +
                (opt.hint ? '<span>' + escapeHtml(opt.hint) + '</span>' : '') + '</span>' +
            '</button>';
          }).join("") +
        '</div>' +
      '</div>';
    }

            { label: "내가 짠 배치표 올리기", hint: "검사와 개선 제안까지", mode: "meal:manual" }

    function handleSeatCellClick(target) {
      const parts = String(target.dataset.seat || "").split(",");

      const c = Number(parts[1]);

        STATE.seatSwapPick = null;
        STATE.seatMap.dirty = true;

    function generateRoomAssignment(allowMixedGenderPairs) {
      if (!STATE.course) return;
      const floors = String(valueOf("roomFloors") || "")
        .split(/[,\s]+/).map(function (v) { return v.replace(/\D/g, ""); }).filter(Boolean);
      if (!floors.length) {
        showToast("사용할 층을 입력해주세요.", "예: 7, 8, 9, 10", "error");
        return;
      }
      STATE.roomConfig = {
        floors: valueOf("roomFloors"),
        perFloor: valueOf("roomsPerFloor")
      };
      const payload = {
        siteId: STATE.course.siteId,
        floors: floors,
        roomsPerFloor: Number(valueOf("roomsPerFloor") || 20),
        priority: valueOf("roomPriority") || "sameOrg",
        mustPairs: collectAgreedRoommatePairs(),
        allowMixedGenderPairs: Boolean(allowMixedGenderPairs),
        commit: false
      };
      showLoader("방 배정 중", "성별로 층을 나누고 짝을 맞추고 있습니다.");
      api("portalGenerateRoomAssignment", payload)
        .then(function (result) {
          if (result.needsConfirm === "mixedGender") {
            confirmMixedGenderPairs(result.conflicts || []);
            return;
          }
          STATE.roomPlan = {
            assignments: result.assignments || [],
            report: result.report || {},
            dirty: true
          };
          STATE.roomSwapPick = null;
          updateCourseTabContent();
          showToast("방을 배정했습니다.", result.message + " 이름을 두 번 눌러 서로 바꿀 수 있습니다.");
        })
        .catch(showErrorToast)
        .finally(hideLoader);
    }

    /** 성별이 다른 요청만 빼고 다시 배정합니다. */
    function generateRoomAssignmentIgnoringMixed() {
      const roster = (STATE.roster && STATE.roster.roster) || [];
      const genderByKey = {};
      roster.forEach(function (r) { genderByKey[r.key] = r.gender; });
      STATE.roomPairFilter = function (pair) {
        return genderByKey[pair[0]] === genderByKey[pair[1]];
      };
      generateRoomAssignment(false);
      STATE.roomPairFilter = null;
    }

    /** 동의된 룸메이트 요청을 '반드시 같은 방' 목록으로 씁니다. */
    function collectAgreedRoommatePairs() {
      const requests = (STATE.roster && STATE.roster.requests) || [];
      const pairs = requests
        .filter(function (r) { return r.status === "동의완료"; })
        .map(function (r) { return [r.fromKey, r.toKey]; });
      return STATE.roomPairFilter ? pairs.filter(STATE.roomPairFilter) : pairs;
    }

    function handleRoomPersonClick(target) {
      const key = target.dataset.roomPerson;
      if (!STATE.roomPlan) return;
      if (!STATE.roomSwapPick) {
        STATE.roomSwapPick = key;
      } else if (STATE.roomSwapPick === key) {
        STATE.roomSwapPick = null;
      } else {
        const list = STATE.roomPlan.assignments;
        const a = list.filter(function (x) { return x.key === STATE.roomSwapPick; })[0];
        const b = list.filter(function (x) { return x.key === key; })[0];
        if (a && b) {
          if (a.gender !== b.gender) {
            showToast("성별이 달라 바꿀 수 없습니다.", a.name + " · " + b.name, "error");
            STATE.roomSwapPick = null;
            updateCourseTabContent();
            return;
          }
          const room = a.room;
          a.room = b.room;
          b.room = room;
          STATE.roomPlan.dirty = true;
        }
        STATE.roomSwapPick = null;
      }
      updateCourseTabContent();
    }

    /* --- 수동 배치표 --- */
    function handleManualSeatingFile(file) {
      if (file.size > 5 * 1024 * 1024) {
        showToast("파일이 너무 큽니다.", "엑셀·CSV 파일은 5MB 이하로 올려주세요.", "error");
        return;
      }
      showLoader("배치표 확인 중", file.name + "을 읽고 있습니다.");
      ensureXlsxLoaded()
        .then(function (XLSX) { return readWorkbookFile(XLSX, file, "roster"); })
        .then(function (workbookData) {
          const rows = parseManualSeatingRows(workbookData.rows);
          if (!rows.length) throw new Error("이름과 테이블 번호를 찾지 못했습니다. 양식을 확인해주세요.");
          const payload = { siteId: STATE.course.siteId, rows: rows, commit: false };
          return api("portalApplyManualSeating", payload).then(function (result) {
            STATE.pendingSeatingPayload = payload;
            STATE.pendingSeatingMode = "manual";
            STATE.manualSeatingRows = rows;
            showSeatingPreview(result, "manual");
            // 저장 여부와 별개로, 바로 검사해서 개선안을 찾아 둡니다.
            return reviewManualSeating(result);
          });
        })
        .catch(showErrorToast)
        .finally(hideLoader);
    }

    /**
     * 이름 + 선택 식별자(소속/휴대폰 뒤 4자리) + 끼니별 테이블 번호.
     * 예전처럼 첫 열 이름, 나머지 테이블 번호인 파일도 그대로 읽습니다.
     */
    function parseManualSeatingRows(rows) {
      const grid = trimEmptyGrid(rows);
      if (!grid.length) return [];
      let headerIndex = -1;
      let nameCol = 0;
      let identityCol = -1;
      let tableCols = [];
      for (let r = 0; r < Math.min(8, grid.length); r++) {
        const normalized = (grid[r] || []).map(function (cell) {
          return String(cell || "").replace(/\s/g, "").toLowerCase();
        });
        const candidateName = normalized.findIndex(function (cell) {
          return /^(이름|성명|성함|교육생|참가자|name)$/.test(cell);
        });
        if (candidateName === -1) continue;
        headerIndex = r;
        nameCol = candidateName;
        identityCol = normalized.findIndex(function (cell, index) {
          return index !== nameCol && /(소속|기관|회사|휴대폰|전화|뒤4|식별|company|phone)/.test(cell);
        });
        tableCols = normalized.map(function (cell, index) {
          return index !== nameCol && index !== identityCol && /(끼니|식사|테이블|점심|저녁|table|meal)/.test(cell)
            ? index : -1;
        }).filter(function (index) { return index >= 0; });
        if (!tableCols.length) {
          tableCols = normalized.map(function (_, index) { return index; })
            .filter(function (index) { return index !== nameCol && index !== identityCol; });
        }
        break;
      }

      if (headerIndex === -1) {
        headerIndex = -1;
        nameCol = 0;
        tableCols = Array.from({ length: Math.max(0, (grid[0] || []).length - 1) }, function (_, index) {
          return index + 1;
        });
      }

      const out = [];
      grid.forEach(function (row, index) {
        if (index <= headerIndex) return;
        const name = String(row[nameCol] || "").trim();
        if (!name) return;
        const discriminator = identityCol >= 0 ? String(row[identityCol] || "").trim() : "";
        const tables = tableCols.map(function (col) {
          return String(row[col] || "").replace(/[^\d]/g, "");
        });
        if (!tables.some(Boolean)) return;
        out.push({
          name: name,
          discriminator: discriminator,
          phoneLast4: /^\d{4}$/.test(discriminator.replace(/\D/g, ""))
            ? discriminator.replace(/\D/g, "") : "",
          organization: /\D/.test(discriminator) ? discriminator : "",
          tables: tables
        });
      });
      return out;
    }

    function downloadManualSeatingTemplate() {
      const slots = (STATE.course && STATE.course.mealSlots) || [];
      const mealHeaders = slots.length ? slots.map(function (slot, index) {
        return Number(slot.day || index + 1) + "일차 " + mealTypeLabel(slot.type) + " 테이블";
      }) : ["1끼니 테이블", "2끼니 테이블", "3끼니 테이블"];
      const header = ["이름", "소속 또는 휴대폰 뒤 4자리"].concat(mealHeaders);
      const sample = [
        ["김벤처", "한국벤처투자", "1", "3", "2"],
        ["홍길동", "1234", "2", "1", "3"],
        ["홍길동", "5678", "3", "2", "1"]
      ];
      const csv = [header].concat(sample)
        .map(function (row) { return row.join(","); }).join("\r\n");
      const blob = new Blob(["\ufeff" + csv], { type: "text/csv;charset=utf-8;" });
      const url = URL.createObjectURL(blob);
      const link = document.createElement("a");
      link.href = url;
      link.download = "자리배치표_양식.csv";
      link.click();
      URL.revokeObjectURL(url);
      showToast("양식을 내려받았습니다.", "동명이인은 두 번째 열에 소속 또는 휴대폰 뒤 4자리를 적어주세요.", "info");
    }

    /* ==========================================================
     * 이미 만들어둔 배치표(엑셀) 읽어오기
     * 좌석표 · 방 배정 · 식사 배치를 시트 이름으로 알아서 구분합니다.
     * ========================================================== */
    function handlePreparedPlanFile(file) {
      if (file.size > 8 * 1024 * 1024) {
        showToast("파일이 너무 큽니다.", "8MB 이하로 올려주세요.", "error");
        return;
      }
      showLoader("배치표 읽는 중", file.name + "의 시트를 확인하고 있습니다.");
      ensureXlsxLoaded()
        .then(function (XLSX) {
          return new Promise(function (resolve, reject) {
            const reader = new FileReader();
            reader.onload = function () {
              try {
                resolve(XLSX.read(reader.result, { type: "array", dense: true, sheetRows: 2001 }));
              } catch (error) { reject(error); }
            };
            reader.onerror = function () { reject(new Error("파일을 읽지 못했습니다.")); };
            reader.readAsArrayBuffer(file);
          });
        })
        .then(function (workbook) {
          const found = detectPreparedSheets(workbook);
          if (!found.seat && !found.room && !found.meal) {
            throw new Error("좌석·방·식사 배치로 볼 만한 시트를 찾지 못했습니다. 시트 이름에 ‘좌석’·‘방’·‘식사’를 넣어주세요.");
          }
          showPreparedImportModal(found);
        })
        .catch(showErrorToast)
        .finally(hideLoader);
    }

    function detectPreparedSheets(workbook) {
      const XLSX = window.XLSX;
      const out = { seat: null, room: null, meal: null };
      workbook.SheetNames.forEach(function (name) {
        const rows = XLSX.utils.sheet_to_json(workbook.Sheets[name], { header: 1, defval: "" });
        const grid = trimEmptyGrid(rows);
        if (!grid.length) return;
        if (!out.seat && /좌석|자리|강의장|seat/i.test(name)) out.seat = { name: name, grid: grid };
        else if (!out.room && /방|객실|룸메|숙박|room/i.test(name)) out.room = { name: name, grid: grid };
        else if (!out.meal && /식사|배치|테이블|meal|table/i.test(name)) out.meal = { name: name, grid: grid };
      });
      return out;
    }

    function applyPreparedPlan() {
      const found = STATE.pendingPreparedPlan || {};
      const tasks = [];
      let roomUpdates = null;

      if (found.room) {
        roomUpdates = parseRoomRows(found.room.grid);
        if (roomUpdates.unresolvedNames && roomUpdates.unresolvedNames.length) {
          showToast(
            "방 배정표에서 연결하지 못한 항목이 있습니다.",
            roomUpdates.unresolvedNames.slice(0, 6).join(", ") +
              (roomUpdates.unresolvedNames.length > 6 ? " 외" : "") +
              ". 동명이인은 소속 또는 휴대폰 뒤 4자리를 이름과 함께 적어주세요.",
            "error"
          );
          return;
        }
      }

      closeModal();

      if (found.seat) {
        STATE.seatMap = buildSeatMapFromPreparedGrid(found.seat.grid);
      }

      if (roomUpdates) {
        if (roomUpdates.length) {
          tasks.push(api("portalSetRosterSeatingFlags", STATE.course.siteId, roomUpdates));
        }
      }

      if (found.meal) {
        const rows = parseManualSeatingRows(found.meal.grid);
        if (rows.length) {
          tasks.push(api("portalApplyManualSeating", {
            siteId: STATE.course.siteId, rows: rows, commit: true
          }));
        }
      }

      showLoader("배치표 반영 중", "명부와 운영 시트에 옮기고 있습니다.");
      Promise.all(tasks)
        .then(function () {
          STATE.pendingPreparedPlan = null;
          showToast("배치표를 반영했습니다.", "좌석표는 아래에서 확인 후 저장해주세요.");
          return loadCourseRoster(true);
        })
        .catch(showErrorToast)
        .finally(hideLoader);
    }

    /** [방번호, 이름, 이름] 또는 [이름, 방번호] 어느 쪽이든 읽습니다. */
    function parseRoomRows(grid) {
      const updates = [];
      const used = {};
      const unresolved = [];
      grid.forEach(function (row, index) {
        const cells = row.map(function (c) { return String(c || "").trim(); }).filter(Boolean);
        if (!cells.length) return;
        if (index === 0 && /방|호수|이름|room/i.test(cells.join(""))) return;
        const roomCell = cells.filter(function (c) { return /^\d{3,4}(호)?$/.test(c); })[0];
        if (!roomCell) {
          unresolved.push(cells.join(" / "));
          return;
        }
        const room = roomCell.replace(/호/g, "");
        cells.forEach(function (cell) {
          if (cell === roomCell) return;
          const person = resolveRosterIdentity(cell, used);
          if (person && person.key) {
            used[person.key] = true;
            updates.push({ key: person.key, room: room });
          } else if (!/^(이름|성명|비고|소속|휴대폰|남|여|남성|여성|1인실|2인실)$/i.test(cell)) {
            unresolved.push(cell);
          }
        });
      });
      updates.unresolvedNames = unresolved.filter(function (value, index, list) {
        return list.indexOf(value) === index;
      });
      return updates;
    }

    function normalizeRosterIdentity(value) {
      return String(value || "").toLowerCase().replace(/[^0-9a-z가-힣]/g, "");
    }

    /** 이름·소속·휴대폰 뒤 4자리를 함께 보고 명부의 한 사람을 찾습니다. */
    function resolveRosterIdentity(value, used) {
      const roster = (STATE.roster && STATE.roster.roster) || [];
      const raw = String(value || "").trim();
      const target = normalizeRosterIdentity(raw);
      if (!target) return null;
      let matches = roster.filter(function (person) {
        return !person.excluded && normalizeRosterIdentity(person.name) === target;
      });
      if (!matches.length) {
        matches = roster.filter(function (person) {
          const name = normalizeRosterIdentity(person.name);
          return !person.excluded && name && (target.indexOf(name) !== -1 || name.indexOf(target) !== -1);
        });
      }
      if (matches.length > 1) {
        const narrowed = matches.filter(function (person) {
          const org = normalizeRosterIdentity(person.organization || person.appliedOrganization || "");
          const last4 = String(person.phoneLast4 || "");
          return (org && target.indexOf(org) !== -1) || (last4 && target.indexOf(last4) !== -1);
        });
        if (narrowed.length) matches = narrowed;
      }
      const available = matches.filter(function (person) { return !(used && used[person.key]); });
      return available.length === 1 ? available[0] : null;
    }

    function isSeatLayoutLabel(value) {
      return /^(무대|stage|교탁|통로|창가|출입구|좌석|자리|강의장|배치도|성명|이름|번호)$/i
        .test(String(value || "").replace(/\s/g, ""));
    }

    /** 수동 좌석 엑셀을 명부 키가 연결된 안전한 격자로 바꿉니다. */
    function buildSeatMapFromPreparedGrid(sourceGrid) {
      const rows = (sourceGrid || []).slice(0, 30);
      const cols = Math.min(30, Math.max.apply(null, rows.map(function (row) { return row.length; })) || 1);
      const used = {};
      const unresolved = [];
      const grid = rows.map(function (row) {
        return Array.from({ length: cols }, function (_, index) {
          const raw = String(row[index] || "").trim();
          if (!raw || isSeatLayoutLabel(raw)) return null;
          const person = resolveRosterIdentity(raw, used);
          if (!person) {
            unresolved.push(raw);
            return { key: "", name: raw, org: "", unresolved: true };
          }
          used[person.key] = true;
          return {
            key: person.key,
            name: person.name,
            org: person.organization || person.appliedOrganization || "",
            phoneLast4: person.phoneLast4 || ""
          };
        });
      });
      const roster = (STATE.roster && STATE.roster.roster) || [];
      const duplicateCounts = {};
      roster.forEach(function (person) {
        const key = normalizeRosterIdentity(person.name);
        duplicateCounts[key] = Number(duplicateCounts[key] || 0) + 1;
      });
      return {
        grid: grid,
        cols: cols,
        rows: grid.length,
        report: {
          cols: cols,
          rows: grid.length,
          seated: Object.keys(used).length,
          emptySeats: Math.max(0, cols * grid.length - Object.keys(used).length),
          duplicateNames: roster.filter(function (person) {
            return duplicateCounts[normalizeRosterIdentity(person.name)] > 1;
          }).map(function (person) { return person.name; }).filter(function (name, index, list) {
            return list.indexOf(name) === index;
          }),
          unresolvedNames: unresolved
        },
        dirty: true
      };
    }

    /* ==========================================================
     * 수동 배치 검증 + 개선 제안
     * "자동을 못 믿겠다"는 분께 자동을 강요하지 않습니다.

          }
        });
    }

    /** 미리보기 모달 안에 검사 결과와 제안을 끼워 넣습니다. */

      const slot = document.getElementById("manualReviewSlot");

      const issues = [
        ["같은 방 룸메이트끼리 같은 테이블", counts.roommate],

    /** 제안을 그대로 받아 확정 대상 rows 를 바꿉니다. */
    function applySeatingSuggestion() {
      const review = STATE.manualReview;
      if (!review || !(review.improvedEntries || []).length) return;
      // 서버가 명부 키와 함께 돌려준 행을 그대로 사용해야 동명이인과 누락 행이
      // 있어도 사람 순서가 뒤섞이지 않습니다.
      const payload = Object.assign({}, STATE.pendingSeatingPayload || {}, {
        rows: review.improvedEntries.map(function (entry) {
          return {
            participantKey: entry.participantKey,
            name: entry.name,
            organization: entry.organization,
            phoneLast4: entry.phoneLast4,
            tables: (entry.tables || []).slice()
          };
        })
      });
      payload.commit = false;
      showLoader("제안 적용 중", "바뀐 배치로 다시 확인합니다.");
      api("portalApplyManualSeating", payload)
        .then(function (result) {
          STATE.pendingSeatingPayload = payload;
          STATE.manualSeatingRows = payload.rows;
          closeModal();
          showSeatingPreview(result, "manual");
          showToast("제안을 반영했습니다.", "확인 후 확정해주세요.", "success");
          return reviewManualSeating(result);
        })
        .catch(showErrorToast)
        .finally(hideLoader);
    }

    function commitSeating(mode) {
      const payload = Object.assign({}, STATE.pendingSeatingPayload || {}, { commit: true });
      if (!payload.siteId) return;
      closeModal();
      showLoader("자리 배정 확정 중", "운영 시트에 저장하고 사용자 사이트에 반영합니다.");
      api(mode === "manual" ? "portalApplyManualSeating" : "portalGenerateSeatingFromRoster", payload)
        .then(function (result) {
          STATE.pendingSeatingPayload = null;
          showToast("자리 배정을 확정했습니다.", result.message);
          if (result.spreadsheetUrl) window.open(result.spreadsheetUrl, "_blank", "noopener");
          return loadCourseRoster(true).then(function () { advanceSeatingStep("meal"); });
        })
        .catch(showErrorToast)
        .finally(hideLoader);
    }

    function getUserPageBlockRuntime(block) {
      if (block.enabled === false) return { status: "hidden", label: "숨김" };
      if (block.scheduleMode === "scheduled") {
        const now = Date.now();
        const start = new Date(block.showFrom || "").getTime();
        const end = new Date(block.showUntil || "").getTime();
        if (isFinite(start) && now < start) return { status: "scheduled", label: "예약" };
        if (isFinite(end) && now >= end) return { status: "expired", label: "종료" };
      }
      if (getUserPageBlockNeedsConnection(block)) return { status: "draft", label: "연결 필요" };
      return { status: "active", label: "공개 중" };
    }

    function getUserPageBlockNeedsConnection(block) {
      if (["custom-link", "open-chat"].indexOf(block.type) !== -1) return !block.url;
      if (block.type === "custom-file") return !block.fileUrl;
      if (block.type === "remember") return !(block.url || (STATE.course.settings && STATE.course.settings.rememberUrl));
      if (block.type === "survey") return !(block.url || (STATE.course.settings && STATE.course.settings.surveyUrl));
      if (block.type === "meal") return !Number(STATE.course.courseDays || 0) || !(STATE.course.mealSlots || []).length;
      if (block.type === "room") return !Number(STATE.course.courseDays || 0) || (STATE.course.settings && STATE.course.settings.lodgingType === "commute");
      if (block.type === "seat") return !Number(STATE.course.courseDays || 0);
      if (block.type === "materials") return !STATE.course.materialFolderId;
      return false;
    }

    function isSafeUserPageHttpsUrl(value) {
      try {
        const parsed = new URL(String(value || ""));
        return parsed.protocol === "https:" && Boolean(parsed.hostname) &&
          !parsed.username && !parsed.password;
      } catch (error) { return false; }
    }

    function submitUserPageBlockForm(form) {
      const type = valueOf("pageBlockType") || "custom-link";
      const meta = getUserPageBlockMeta(type);
      const title = valueOf("pageBlockTitle");
      const url = valueOf("pageBlockUrl");
      const scheduled = checkedOf("pageBlockScheduled");
      const showFrom = valueOf("pageBlockShowFrom");
      const showUntil = valueOf("pageBlockShowUntil");
      const errorBox = document.getElementById("pageBlockFormError");
      const fail = function (message) {
        if (errorBox) { errorBox.textContent = message; errorBox.classList.remove("hidden"); }
      };
      if (!title) return fail("사용자 화면에 보일 제목을 입력해주세요.");
      if (["custom-link", "open-chat"].indexOf(type) !== -1 && !url) return fail("연결 주소를 입력해주세요.");
      if (url && !isSafeUserPageHttpsUrl(url)) return fail("연결 주소는 로그인 정보가 포함되지 않은 올바른 https:// 주소여야 합니다.");
      if (scheduled && (!showFrom || !showUntil)) return fail("공개 시작과 종료 시각을 모두 입력해주세요.");
      if (scheduled && new Date(showFrom).getTime() >= new Date(showUntil).getTime()) return fail("공개 종료 시각은 시작 시각보다 뒤여야 합니다.");
      const editIndex = Number(form.dataset.editIndex);
      const current = editIndex >= 0 ? (STATE.pageBlocks[editIndex] || {}) : {};
      const block = {
        id: current.id || "PB-" + Date.now().toString(36) + "-" + Math.random().toString(36).slice(2, 8),
        type: type,
        title: title,
        description: valueOf("pageBlockDescription"),
        buttonLabel: valueOf("pageBlockButtonLabel") || meta.button,
        url: url,
        fileId: type === "custom-file" ? (current.fileId || "") : "",
        fileName: type === "custom-file" ? (current.fileName || "") : "",
        fileUrl: type === "custom-file" ? (current.fileUrl || "") : "",
        enabled: checkedOf("pageBlockEnabled"),
        scheduleMode: scheduled ? "scheduled" : "always",
        showFrom: scheduled ? localInputToIso(showFrom) : "",
        showUntil: scheduled ? localInputToIso(showUntil) : ""
      };
      const finish = function () {
        if (editIndex >= 0) STATE.pageBlocks.splice(editIndex, 1, block);
        else STATE.pageBlocks.push(block);
        STATE.pageBuilderDirty = true;
        STATE.pendingPageBlockFile = null;
        closeModal(true);
        updateCourseTabContent();
        showToast(editIndex >= 0 ? "카드를 수정했습니다." : "카드를 화면에 추가했습니다.", "오른쪽 미리보기에서 확인한 뒤 실제 사이트에 반영해주세요.");
      };
      if (type !== "custom-file" || !STATE.pendingPageBlockFile) {
        if (type === "custom-file" && block.enabled && !block.fileUrl) return fail("사용자에게 보여줄 파일을 선택해주세요.");
        finish();
        return;
      }
      const file = STATE.pendingPageBlockFile;
      if (file.size > 8 * 1024 * 1024) return fail("파일은 8MB 이하만 올릴 수 있습니다.");
      setModalBusy("userPageBlockForm", true, "파일 올리는 중…");
      showLoader("안내 파일 업로드 중", "안전한 형식인지 확인한 뒤 교육 자료 폴더에 저장하고 있습니다.");
      readFileAsBase64(file)
        .then(function (payloadFile) {
          return api("portalUploadCourseFiles", {
            siteId: STATE.course.siteId,
            type: "material",
            requestId: "page_block_" + STATE.course.siteId + "_" + file.name + "_" + file.size + "_" + file.lastModified,
            files: [payloadFile]
          });
        })
        .then(function (result) {
          const uploaded = (result.uploadedFiles || [])[0];
          if (!uploaded) throw new Error("업로드 결과에서 파일을 확인하지 못했습니다.");
          block.fileId = uploaded.id || "";
          block.fileName = uploaded.name || file.name;
          block.fileUrl = uploaded.viewUrl || uploaded.downloadUrl || "";
          STATE.resourceLoaded.materials = false;
          finish();
        })
        .catch(function (error) { fail(error.message); })
        .finally(function () { setModalBusy("userPageBlockForm", false); hideLoader(); });
    }

    function moveUserPageBlock(index, direction) {
      const from = Number(index);
      const to = from + (direction === "up" ? -1 : 1);
      if (from < 0 || to < 0 || from >= STATE.pageBlocks.length || to >= STATE.pageBlocks.length) return;
      const item = STATE.pageBlocks.splice(from, 1)[0];
      STATE.pageBlocks.splice(to, 0, item);
      STATE.pageBuilderDirty = true;
      updateCourseTabContent();
    }

    }

    function renderSegmentMeals(segment, index, dayCount) {

      for (let day = 1; day <= dayCount; day++) {

            return Number(meal.day) === day && meal.type === type;
          });
          tiles += '<label class="segment-meal' + (checked ? " is-on" : "") + '">' +
            '<input type="checkbox" data-segment-meal="' + index + ':' + day + ':' + type + '"' +
              (checked ? ' checked' : '') + '>' +
            '<span>' + day + '일차 ' + mealTypeLabel(type) + '</span></label>';
        });
      }
      return '<div class="segment-meals"><span class="segment-meals-label">식사 자리 배치가 필요한 끼니</span>' +
        '<div class="segment-meal-tiles">' + tiles + '</div></div>';
    }

    function countSegmentDays(segment) {
      const start = new Date(String(segment.startDate || "") + "T00:00:00");
      const end = new Date(String(segment.endDate || segment.startDate || "") + "T00:00:00");
      if (isNaN(start.getTime()) || isNaN(end.getTime())) return 0;
      return Math.max(1, Math.round((end - start) / 86400000) + 1);
    }

    function getCourseDateBounds() {
      const dates = (STATE.course && STATE.course.dates || [])
        .map(koreanDateTextToIso)
        .filter(Boolean)
        .sort();
      if (!dates.length) return null;
      return { first: dates[0], last: dates[dates.length - 1] };
    }

    function shiftIsoDate(iso, offsetDays) {
      const parts = String(iso || "").split("-").map(Number);
      if (parts.length !== 3) return "";
      const date = new Date(parts[0], parts[1] - 1, parts[2]);
      date.setDate(date.getDate() + Number(offsetDays || 0));
      return date.getFullYear() + "-" + pad(date.getMonth() + 1) + "-" + pad(date.getDate());
    }

    function buildSchedulePlan() {
      const bounds = getCourseDateBounds();
      if (!bounds) return null;
      const plan = SCHEDULE_PRESET.map(function (item) {
        const base = item[1] === "last" ? bounds.last : bounds.first;
        const day = shiftIsoDate(base, item[2]);
        return { id: item[0], label: item[4], value: day + "T" + item[3] };
      });
      // 결제 마감은 날짜만 받는 칸입니다.
      plan.push({
        id: "settingsPaymentDueAt",
        label: "결제 마감일",
        value: shiftIsoDate(bounds.first, -14),
        dateOnly: true
      });
      return plan;
    }

    function formatPlanValue(item) {
      const value = String(item.value || "");
      if (item.dateOnly) return value.replace(/-/g, ". ") + ".";
      const parts = value.split("T");
      return parts[0].replace(/-/g, ". ") + ". " + (parts[1] || "");
    }

    function saveCreateDraft() {
      const form = document.getElementById("courseCreateForm");
      if (!form) return;
      const values = {};
      form.querySelectorAll("input[id], select[id]").forEach(function (input) {
        if (input.id === "courseAccessCode" || input.type === "file") return;
        values[input.id] = input.type === "checkbox" ? input.checked : input.value;
      });
      const courseType = form.querySelector('[name="courseType"]:checked');
      const meals = {};
      form.querySelectorAll(".meal-checkbox").forEach(function (input) { meals[input.value] = input.checked; });
      try {
        sessionStorage.setItem(DRAFT_KEY, JSON.stringify({
          savedAt: Date.now(),
          requestId: STATE.createRequestId,
          step: STATE.createCourseStep,
          courseType: courseType ? courseType.value : "vup",
          contacts: collectCourseContacts(true),
          values: values,
          meals: meals
        }));
        const draftStatus = document.getElementById("createDraftStatus");
        if (draftStatus) draftStatus.innerHTML = "<i></i> 방금 저장됨";
      } catch (error) {
        const draftStatus = document.getElementById("createDraftStatus");
        if (draftStatus) draftStatus.textContent = "이 브라우저에서는 자동 저장을 사용할 수 없습니다.";
      }
    }

    function restoreCreateDraft() {
      let draft = null;
      try { draft = JSON.parse(sessionStorage.getItem(DRAFT_KEY) || "null"); }
      catch (error) { draft = null; }
      if (!draft || Date.now() - Number(draft.savedAt || 0) > 24 * 60 * 60 * 1000) return;
      STATE.createRequestId = String(draft.requestId || STATE.createRequestId || createRequestId());
      const type = document.querySelector('[name="courseType"][value="' + (draft.courseType === "general" ? "general" : "vup") + '"]');
      if (type) { type.checked = true; toggleCreateCourseType(type.value); }
      if (Array.isArray(draft.contacts) && draft.contacts.length) {
        renderCourseContactRows(draft.contacts);
      }
      if (draft.values && draft.values.courseDays) {
        const days = document.getElementById("courseDays");
        if (days) { days.value = draft.values.courseDays; renderCreateDateAndMealFields(); }
      }
      Object.keys(draft.values || {}).forEach(function (id) {
        const input = document.getElementById(id);
        if (!input || id === "courseAccessCode") return;
        if (input.type === "checkbox") input.checked = Boolean(draft.values[id]);
        else input.value = draft.values[id];
      });
      // 날짜는 년·월·일 칸과 숨은 date 입력 두 곳에 있으므로, 복구 후 서로 맞춰줍니다.
      const restoredDays = clamp(Number(valueOf("courseDays") || 3), 1, 5);
      for (let day = 1; day <= restoredDays; day++) {
        if (valueOf("courseDateY" + day)) syncKoreanDateToHidden(day);
        else syncHiddenToKoreanDate(day);
      }
      refreshMealTileLabels();
      document.querySelectorAll(".meal-checkbox").forEach(function (input) {
        if (draft.meals && draft.meals[input.value] != null) input.checked = Boolean(draft.meals[input.value]);
      });
      // '식사 배치 불필요'가 켜져 있었다면 타일도 다시 비활성화합니다.
      const skipBox = document.getElementById("courseMealSkip");
      const mealRoot = document.getElementById("courseMealSlots");
      if (skipBox && mealRoot) mealRoot.classList.toggle("meal-slot-disabled", skipBox.checked);
      STATE.createCourseStep = Math.max(1, Math.min(4, Number(draft.step || 1)));
      updateMealSelectionSummary();
      updateCreateSummary();
      updateCreateWizard();
      showToast("작성 중이던 강의를 복구했습니다.", "24시간 이내의 입력값과 생성 요청 번호를 복원했습니다.");
    }

    function createCourse(form) {
      if (STATE.createSubmitting) return;
      for (let step = 1; step <= 4; step++) {
        if (!validateCreateStep(step)) {
          STATE.createCourseStep = step;
          updateCreateWizard();
          return;
        }
      }
      const formData = new FormData(form);
      const isGeneral = formData.get("courseType") === "general";
      const days = Number(valueOf("courseDays"));
      const dates = [];
      for (let day = 1; day <= days; day++) {
        const dateValue = valueOf("courseDate" + day);
        if (!dateValue) {
          showToast("교육 날짜를 확인해주세요.", day + "일차 날짜가 비어 있습니다.", "error");
          return;
        }
        dates.push(dateValue);
      }
      const mealSkipped = checkedOf("courseMealSkip");
      const mealSlots = mealSkipped ? [] : collectMealSlots(document);
      if (!mealSkipped && !mealSlots.length) {
        STATE.createCourseStep = 3;
        updateCreateWizard();
        showToast("식사 일정을 선택해주세요.", "사용자에게 안내할 식사를 하나 이상 선택해야 합니다.", "error");
        return;
      }
      const educationValue = isGeneral ? valueOf("courseTitle") : valueOf("courseGeneration");
      if (!educationValue) {
        STATE.createCourseStep = 1;
        updateCreateWizard();
        setFieldError(isGeneral ? "courseTitle" : "courseGeneration",
          isGeneral ? "교육명을 입력해주세요" : "기수를 입력해주세요");
        return;
      }
      const contacts = collectCourseContacts(false).filter(function (contact) {
        return contact.name && contact.phone;
      });
      const primaryContact = contacts.find(function (contact) { return contact.primary; }) || contacts[0];
      const secondaryContact = contacts.find(function (contact) { return contact !== primaryContact; }) || {};
      const payload = {
        isNonVup: isGeneral,
        generation: educationValue,
        managerName: primaryContact.name,
        managerPhone: primaryContact.phone,
        subManagerName: secondaryContact.name || "",
        subManagerPhone: secondaryContact.phone || "",
        courseDays: days,
        lodgingType: (document.querySelector('input[name="lodgingType"]:checked') || {}).value || "stay",
        mealSkipped: mealSkipped,
        mealSlots: mealSlots,
        requestId: STATE.createRequestId,
        settings: {
          accessCode: valueOf("courseAccessCode"),
          rememberUrl: valueOf("courseRememberUrl"),
          rememberEnabled: Boolean(valueOf("courseRememberUrl")),
          busInfo: "",
          progressText: "교육 준비 중",
          organizationEditEnabled: true,
          subManagerName: secondaryContact.name || "",
          subManagerPhone: secondaryContact.phone || "",
          contactPersons: contacts.map(function (contact) {
            return { name: contact.name, phone: contact.phone, primary: contact === primaryContact };
          }),
          targetAudience: valueOf("courseTargetAudience"),
          capacity: valueOf("courseCapacity"),
          courseFee: valueOf("courseFee"),
          applicationDueDate: valueOf("courseApplyDueDate"),
          sourceDocumentName: String(STATE.courseDocumentFileName || ""),
          paymentDueAt: valueOf("coursePaymentDueAt"),
          lodgingType: (document.querySelector('input[name="lodgingType"]:checked') || {}).value || "stay",
          mealSkipped: mealSkipped,
          defaultVenue: valueOf("courseVenue"),
          defaultStartTime: valueOf("courseStartTime"),

    function uploadSelectedFiles(type) {
      const files = STATE.selectedFiles[type] || [];
      if (!files.length) return;
      if (files.length > 10) {
        showToast("파일이 너무 많습니다.", "한 번에 최대 10개까지 올릴 수 있습니다.", "error");
        return;
      }
      const tooLarge = files.find(function (file) { return file.size > 8 * 1024 * 1024; });
      if (tooLarge) {
        showToast("파일 용량을 확인해주세요.", tooLarge.name + " 파일이 8MB를 초과합니다.", "error");
        return;
      }
      const totalBytes = files.reduce(function (sum, file) { return sum + file.size; }, 0);
      if (totalBytes > 25 * 1024 * 1024) {
        showToast("한 번에 올릴 용량이 큽니다.", "총 25MB 이하로 나누어 올려주세요.", "error");
        return;
      }
      showLoader(type === "photo" ? "현장 사진 게시 중" : "교재 게시 중", "사용자 사이트에 바로 반영하고 있습니다.");
      const failed = [];
      let uploadedCount = 0;
      files.reduce(function (chain, file) {
        return chain.then(function () {
          return readFileAsBase64(file)
            .then(function (payloadFile) {
              return api("portalUploadCourseFiles", {
                siteId: STATE.course.siteId,
                type: type,
                requestId: "upload_" + STATE.course.siteId + "_" + file.name + "_" + file.size + "_" + file.lastModified,
                files: [payloadFile]
              });
            })
            .then(function () { uploadedCount++; })
            .catch(function (error) { failed.push({ file: file, message: error.message }); });
        });
      }, Promise.resolve())
        .then(function () {
          showToast(
            failed.length ? "일부 파일을 게시하지 못했습니다." : (type === "photo" ? "사진을 게시했습니다." : "교재를 게시했습니다."),
            uploadedCount + "개 성공" + (failed.length ? " · " + failed.length + "개 실패 — 실패 파일만 다시 시도할 수 있습니다." : ""),
            failed.length ? "warning" : ""
          );
          return openCourse(STATE.course.siteId, type === "photo" ? "photos" : "materials")
            .then(function (result) {
              if (failed.length) {
                STATE.selectedFiles[type] = failed.map(function (item) { return item.file; });
                handleUploadSelection(type, STATE.selectedFiles[type]);
              }
              return result;
            });
        })
        .catch(showErrorToast)
        .finally(hideLoader);
    }

    function readFileAsBase64(file) {
        return new Promise(function (resolve, reject) {
          const reader = new FileReader();
          reader.onload = function () {
            const result = String(reader.result || "");
            resolve({
              fileName: file.name,
              mimeType: file.type || guessMimeType(file.name),
              base64: result.indexOf(",") === -1 ? result : result.split(",")[1]
            });
          };
          reader.onerror = function () { reject(new Error(file.name + " 파일을 읽지 못했습니다.")); };
          reader.readAsDataURL(file);
        });
    }

    function ensureXlsxLoaded() {
      if (window.XLSX) return Promise.resolve(window.XLSX);
      if (STATE.xlsxPromise) return STATE.xlsxPromise;
      STATE.xlsxPromise = new Promise(function (resolve, reject) {
        const script = document.createElement("script");
        script.src = XLSX_URL;
        script.async = true;
        script.integrity = XLSX_INTEGRITY;
        script.crossOrigin = "anonymous";
        const timeout = window.setTimeout(function () {
          STATE.xlsxPromise = null;
          script.remove();
          reject(new Error("엑셀 처리 도구 응답이 지연되고 있습니다. 네트워크를 확인한 뒤 다시 시도해주세요."));
        }, 12000);
        script.onload = function () {
          window.clearTimeout(timeout);
          if (window.XLSX) resolve(window.XLSX);
          else reject(new Error("엑셀 처리 도구를 불러오지 못했습니다."));
        };
        script.onerror = function () {
          window.clearTimeout(timeout);
          STATE.xlsxPromise = null;
          reject(new Error("엑셀 처리 도구를 불러오지 못했습니다. 네트워크 연결을 확인해주세요."));
        };
        document.head.appendChild(script);
      });
      return STATE.xlsxPromise;
    }

    function handleMealFile(kind, file) {
      if (!file) return;

    function readWorkbookFile(XLSX, file, kind) {
      return new Promise(function (resolve, reject) {
        const reader = new FileReader();
        reader.onload = function () {
          try {
            const workbook = XLSX.read(reader.result, { type: "array", cellDates: false, dense: true, sheetRows: 2001 });
            if (workbook.SheetNames.length > 20) throw new Error("시트 수가 20개를 초과합니다.");
            const keywords = {
              roster: ["교육생", "명단", "참가자", "participant"],
              seat: ["강의장", "좌석", "자리", "seat"],
              roommate: ["룸메이트", "객실", "room"]
            }[kind] || [];
            const sheetName = workbook.SheetNames.find(function (name) {
              const normalized = normalizeHeader(name);
              return keywords.some(function (keyword) {
                return normalized.indexOf(normalizeHeader(keyword)) !== -1;
              });
            }) || workbook.SheetNames[0];
            const rows = XLSX.utils.sheet_to_json(workbook.Sheets[sheetName], {
              header: 1,
              defval: "",
              raw: false,
              blankrows: false
            });
            if (rows.length > 2000) throw new Error("행 수가 2,000개를 초과합니다.");
            const maxColumns = rows.reduce(function (max, row) { return Math.max(max, (row || []).length); }, 0);
            if (maxColumns > 100 || rows.length * maxColumns > 100000) {
              throw new Error("표 크기가 허용 범위를 초과합니다.");
            }
            resolve({ sheetName: sheetName, rows: rows });
          } catch (error) {
            reject(new Error(error && error.message ? error.message : "엑셀 파일을 읽지 못했습니다. 첫 번째 시트의 표 형식을 확인해주세요."));
          }
        };
        reader.onerror = function () { reject(new Error("파일을 읽지 못했습니다.")); };
        reader.readAsArrayBuffer(file);
      });
    }

    function parseRosterRows(rows) {
      const source = Array.isArray(rows) ? rows : [];
      const headerCandidates = {
        name: ["이름", "성명", "교육생명", "참가자명", "name"],
        company: ["회사명", "소속", "기관명", "소속기관", "company", "organization"],
        position: ["직급", "직위", "직책", "position", "title"],
        phone: ["휴대폰번호", "휴대폰", "전화번호", "연락처", "phone", "mobile"]
      };
      let headerIndex = -1;
      let mapping = null;
      for (let rowIndex = 0; rowIndex < Math.min(source.length, 12); rowIndex++) {
        const normalized = (source[rowIndex] || []).map(normalizeHeader);
        const candidate = {};
        Object.keys(headerCandidates).forEach(function (field) {
          candidate[field] = normalized.findIndex(function (value) {
            return headerCandidates[field].some(function (name) { return value === normalizeHeader(name) || value.indexOf(normalizeHeader(name)) !== -1; });
          });
        });
        if (candidate.name !== -1) {
          headerIndex = rowIndex;
          mapping = candidate;
          break;
        }
      }
      if (headerIndex === -1) {
        headerIndex = 0;
        mapping = { name: 0, company: 1, position: 2, phone: 3 };
      }
      const participants = source.slice(headerIndex + 1).map(function (row) {
        return {
          name: String(row[mapping.name] || "").trim(),
          company: mapping.company >= 0 ? String(row[mapping.company] || "").trim() : "",
          position: mapping.position >= 0 ? String(row[mapping.position] || "").trim() : "",
          phone: mapping.phone >= 0 ? String(row[mapping.phone] || "").trim() : ""
        };
      }).filter(function (person) { return person.name; });
      if (!participants.length) throw new Error("교육생 이름을 찾지 못했습니다. 이름 열이 포함된 명단인지 확인해주세요.");
      return participants;
    }

    function trimEmptyGrid(rows) {
      return (Array.isArray(rows) ? rows : [])
        .map(function (row) { return (Array.isArray(row) ? row : []).map(function (cell) { return String(cell == null ? "" : cell).trim(); }); })
        .filter(function (row) { return row.some(Boolean); });
    }

    /** 지난 강의의 운영 설정을 가져오되 새 일정은 관리자가 확인해서 입력합니다. */
    function openDuplicateCourseModal(sourceSiteId) {
      const source = (STATE.courses || []).find(function (course) {
        return String(course.siteId) === String(sourceSiteId);
      });
      if (!source) {
        showToast("복제할 강의를 찾지 못했습니다.", "강의 목록을 새로고침한 뒤 다시 시도해주세요.", "error");
        return;
      }
      STATE.pendingDuplicateSourceId = source.siteId;
      const days = Math.max(1, Math.min(5, Number(source.courseDays || source.days || 1)));
      const dateFields = Array.from({ length: 5 }, function (_, index) {
        const day = index + 1;
        return '<label class="form-field duplicate-date-field" data-duplicate-date-field="' + day + '"' +
          (day > days ? ' hidden' : '') + '><span>' + day + '일차 날짜 <b>*</b></span>' +
          '<input id="duplicateDate' + day + '" type="date"' + (day <= days ? ' required' : '') + '></label>';
      }).join("");
      const body = '<form id="duplicateCourseForm" class="settings-stack" onsubmit="return false">' +
          '<div class="info-callout"><strong>' + escapeHtml(source.title || source.generation || "선택한 강의") + '</strong>' +
            '<span>운영 문구·정보 수정 허용·룸메이트 요청 설정을 가져옵니다. 교육생 명부와 지난 날짜는 복사하지 않습니다.</span></div>' +
          '<div class="form-grid">' +
            '<label class="form-field"><span>새 교육명 <b>*</b></span><input id="duplicateTitle" value="' +
              escapeAttr(source.title || "") + '" maxlength="120" required></label>' +
            '<label class="form-field"><span>새 기수</span><input id="duplicateGeneration" value="' +
              escapeAttr(source.generation || "") + '" maxlength="30" placeholder="예: 19기"></label>' +
            '<label class="form-field"><span>교육 일수 <b>*</b></span><select id="duplicateCourseDays">' +
              [1, 2, 3, 4, 5].map(function (value) {
                return '<option value="' + value + '"' + (value === days ? ' selected' : '') + '>' + value + '일</option>';
              }).join("") + '</select></label>' +
          '</div>' +
          '<div class="form-grid duplicate-date-grid">' + dateFields + '</div>' +
          '<p id="duplicateCourseResult" class="form-error hidden" role="alert"></p>' +
        '</form>';
      showModal(
        "지난 강의를 복제해 새로 만들기",
        "반복 교육을 빠르게 개설하고 새 날짜만 확인합니다.",
        body,
        '<button type="button" class="btn btn-secondary" data-modal-close>취소</button>' +
          '<button type="button" class="btn btn-primary" data-action="run-duplicate-course">새 강의 만들기</button>',
        true,
        "duplicate-course-modal"
      );
      const daysSelect = document.getElementById("duplicateCourseDays");
      if (daysSelect) {
        daysSelect.addEventListener("change", function () {
          const count = Number(daysSelect.value || 1);
          ELEMENTS.modalRoot.querySelectorAll("[data-duplicate-date-field]").forEach(function (field) {
            const visible = Number(field.dataset.duplicateDateField) <= count;
            field.hidden = !visible;
            const input = field.querySelector("input");
            if (input) input.required = visible;
          });
        });
      }
    }

    function runDuplicateCourse() {
      const sourceSiteId = STATE.pendingDuplicateSourceId;
      const title = valueOf("duplicateTitle");
      const days = Number(valueOf("duplicateCourseDays") || 1);
      const resultBox = document.getElementById("duplicateCourseResult");
      if (!sourceSiteId || !title || days < 1 || days > 5) {
        if (resultBox) {
          resultBox.textContent = "교육명과 교육 일수를 확인해주세요.";
          resultBox.classList.remove("hidden");
        }ㅁ
        return;
      }
      const payload = {
        sourceSiteId: sourceSiteId,
        requestId: createRequestId(),
        title: title,
        generation: valueOf("duplicateGeneration"),
        courseDays: days,
        managerName: (STATE.profile && STATE.profile.name) || "",
        managerPhone: (STATE.profile && (STATE.profile.landline || STATE.profile.mobile)) || ""
      };
      for (let day = 1; day <= days; day++) {
        const date = valueOf("duplicateDate" + day);
        if (!date) {
          if (resultBox) {
            resultBox.textContent = day + "일차 날짜를 입력해주세요.";
            resultBox.classList.remove("hidden");
          }
          const field = document.getElementById("duplicateDate" + day);
          if (field) field.focus();
          return;
        }
        payload["day" + day + "Date"] = date;
      }
      const actionButton = ELEMENTS.modalRoot.querySelector('[data-action="run-duplicate-course"]');
      if (actionButton) {
        actionButton.disabled = true;
        actionButton.dataset.idleLabel = actionButton.innerHTML;
        actionButton.innerHTML = '<span class="btn-spinner" aria-hidden="true"></span><span>새 강의를 만드는 중...</span>';
      }
      showLoader("지난 강의 복제 중", "운영 설정을 안전하게 가져오고 새 관리 공간을 만들고 있습니다.");
      api("portalDuplicateCourse", payload)
        .then(function (result) {
          STATE.pendingDuplicateSourceId = null;
          STATE.courseListCache = {};
          STATE.dashboardFetchedAt = 0;
          closeModal(true);
          showToast("새 강의를 만들었습니다.", result.message);
          return openCourse(result.siteId, "overview");
        })
        .catch(function (error) {
          if (resultBox) {
            resultBox.textContent = error.message;
            resultBox.classList.remove("hidden");
          }
        })
        .finally(function () {
          if (actionButton && document.contains(actionButton)) {
            actionButton.disabled = false;
            actionButton.innerHTML = actionButton.dataset.idleLabel || "새 강의 만들기";
          }
          hideLoader();
        });
    }
